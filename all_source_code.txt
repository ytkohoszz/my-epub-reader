  async initAndMigrateLocationsCache() {
    const adapter = this.app.vault.adapter;
    const indexPath = this.getLocationsIndexPath();

    try {
      if (await adapter.exists(indexPath)) {
        const raw = await adapter.read(indexPath);
        const parsed: unknown = JSON.parse(raw);
        if (parsed && typeof parsed === "object") {
          this.locationsIndex = parsed as Record<string, LocationIndexMeta>;
        }
      }
    } catch {
      this.locationsIndex = {};
    }

    const legacy = this.settings.locationsCache;
    if (legacy && typeof legacy === "object" && Object.keys(legacy).length > 0) {
      const pluginDir = this.getPluginDir();
      for (const [bookPath, entry] of Object.entries(legacy)) {
        if (
          entry &&
          typeof entry.sig === "string" &&
          typeof entry.locations === "string" &&
          entry.locations.length > 2
        ) {
          const fileName = hashBookPathToFileName(bookPath);
          const filePath = `${pluginDir}/${fileName}`;
          try {
            await adapter.write(filePath, entry.locations);
            this.locationsIndex[bookPath] = {
              fileName,
              sig: entry.sig,
              updatedAt: entry.updatedAt || Date.now(),
            };
          } catch {
            // Игнорируем
          }
        }
      }
      delete this.settings.locationsCache;
      await this.pruneLocationsCacheFiles();
      await this.saveLocationsIndexToDisk();
      await this.saveData(this.settings);
    } else if ("locationsCache" in this.settings) {
      delete this.settings.locationsCache;
      await this.saveData(this.settings);
    }
  }

  private async saveLocationsIndexToDisk() {
    try {
      await this.app.vault.adapter.write(

  async loadLocationsCacheFile(bookPath: string, expectedSig: string): Promise<string | null> {
    const meta = this.locationsIndex[bookPath];
    if (!meta || meta.sig !== expectedSig || !meta.fileName) return null;

    const fullPath = `${this.getPluginDir()}/${meta.fileName}`;
    try {
      const adapter = this.app.vault.adapter;
      if (!(await adapter.exists(fullPath))) return null;
      const content = await adapter.read(fullPath);
      if (!content || content.length <= 2) return null;
      meta.updatedAt = Date.now();
      void this.saveLocationsIndexToDisk();
      return content;
    } catch {
      return null;
    }
  }

  async saveLocationsCacheFile(bookPath: string, sig: string, serializedLocations: string) {
    const fileName = hashBookPathToFileName(bookPath);
    const fullPath = `${this.getPluginDir()}/${fileName}`;
    try {
      await this.app.vault.adapter.write(fullPath, serializedLocations);
      this.locationsIndex[bookPath] = {
        fileName,
        sig,
        updatedAt: Date.now(),
      };
      await this.pruneLocationsCacheFiles();
      await this.saveLocationsIndexToDisk();
    } catch {
      // Игнорируем
    }
  }

  private async pruneLocationsCacheFiles() {
    const keys = Object.keys(this.locationsIndex);
    if (keys.length <= MAX_CACHED_BOOKS) return;

    keys.sort(
      (a, b) =>
        (this.locationsIndex[a]?.updatedAt ?? 0) - (this.locationsIndex[b]?.updatedAt ?? 0)
    );
    const toRemove = keys.slice(0, keys.length - MAX_CACHED_BOOKS);
    const adapter = this.app.vault.adapter;
    const pluginDir = this.getPluginDir();

    for (const k of toRemove) {
      const meta = this.locationsIndex[k];
      delete this.locationsIndex[k];
      if (meta?.fileName) {
        const fullPath = `${pluginDir}/${meta.fileName}`;
        try {
          if (await adapter.exists(fullPath)) {
            await adapter.remove(fullPath);
          }
        } catch {
          // Игнорируем
        }
      }
    }
  }

  async clearAllLocationsCacheFiles() {
    const adapter = this.app.vault.adapter;
    const pluginDir = this.getPluginDir();
    for (const meta of Object.values(this.locationsIndex)) {
      if (meta?.fileName) {
        const fullPath = `${pluginDir}/${meta.fileName}`;
        try {
          if (await adapter.exists(fullPath)) {
            await adapter.remove(fullPath);

    const clean = epubLinkOrPath.replace(/^\[\[|\]\]$/g, "").split("|")[0]?.split("#")[0]?.trim() || "";
    if (!clean) return;

    let targetFile: TFile | null = null;
    const direct = this.app.vault.getAbstractFileByPath(clean);
    if (direct instanceof TFile) {
      targetFile = direct;
    } else {
      const dest = this.app.metadataCache.getFirstLinkpathDest(clean, "");
      if (dest instanceof TFile) targetFile = dest;
    }

    if (!targetFile) {
      new Notice(`Файл книги «${clean}» не найден в хранилище.`);
      return;
    }

    await this.reloadSettingsFromDisk();

    const navCfi = cfi && cfi.includes(",") ? cfi.replace(/,[^,)]+\)$/, ")") : cfi;
    if (navCfi) {
      this.setLocalBookProgress(targetFile.path, navCfi, new Date().toISOString());
    }

    const existingLeaves = this.app.workspace.getLeavesOfType(VIEW_TYPE);
    for (const leaf of existingLeaves) {
      const view = leaf.view;
      if (view instanceof EpubView && view.file?.path === targetFile.path) {
        if (navCfi) view.ExplicitCfiJumpPending = true;
        this.app.workspace.setActiveLeaf(leaf, { focus: true });
        if (navCfi) await view.jumpToCfi(navCfi, true);
        return;
      }
    }

    const leaf = this.app.workspace.getLeaf(false);
    await leaf.openFile(targetFile);
    if (navCfi && leaf.view instanceof EpubView) {
      await leaf.view.jumpToCfi(navCfi, false);
    }
  }

  refreshAllViews() {
    const leaves = this.app.workspace.getLeavesOfType(VIEW_TYPE);
    for (const leaf of leaves) {
      const view = leaf.view;
      if (view instanceof EpubView) {
        view.updateThemeBtnText();
        view.applyBarPosition();
        view.applySpreadMode(this.settings.spreadMode ?? "none");
        view.applyDesktopWidth(true);
        view.applyStylesAndRestore();
      }
    }
  }

  requestSaveSettings() {
    this.dirty = true;
    if (this.saveTimer) window.clearTimeout(this.saveTimer);
    this.saveTimer = window.setTimeout(() => {
      this.saveTimer = null;
      this.dirty = false;
      void this.saveData(this.settings);
    }, 400);
  }

  async flushSaveSettings() {
    if (this.saveTimer) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
    }
    if (this.dirty) {
      this.dirty = false;
      await this.saveData(this.settings);
    }
  }

  saveSettings() {
    this.requestSaveSettings();
  }

  onunload() {
    document.body.classList.remove("my-epub-fullscreen", "my-epub-sepia-bg", "my-epub-night-bg");
    if (this.saveTimer) {
      window.clearTimeout(this.saveTimer);
      this.saveTimer = null;
    }
    if (this.dirty) {
      this.dirty = false;
      void this.saveData(this.settings);
    }
  }
}

const decodeInline = (s: string) =>
  String(s ?? "").replace(/⟩/g, ")").replace(/⟨/g, "(").trim();

function extractLinkCfi(raw: string): string | null {
  const str = decodeInline(String(raw ?? "")).trim();
  if (!str.startsWith("[[") || !str.endsWith("]]")) return null;
  const inner = str.slice(2, -2);
  const hashIdx = inner.indexOf("#");
  if (hashIdx === -1) return null;
  const barIdx = inner.indexOf("|");
  const endIdx = barIdx !== -1 && barIdx > hashIdx ? barIdx : inner.length;
  let anchor = inner.slice(hashIdx + 1, endIdx).trim();
  if (anchor.startsWith("cfi=")) anchor = anchor.slice(4).trim();
  return anchor.startsWith("epubcfi(") ? anchor : null;
}

function isSameBookTarget(targetStr: string | null, book: BookNoteInfo): boolean {
  if (!targetStr) return false;
  const clean = normStr(targetStr.replace(/\.md$/i, ""));
  return (
    clean === normStr(book.fileName) ||
    clean === normStr(book.title) ||
    normStr(targetStr) === normStr(book.path)
  );
}

function resolveVaultFile(app: App, rawPath: string): TFile | null {
  const clean = rawPath.trim();
  const direct = app.vault.getAbstractFileByPath(clean);
  if (direct instanceof TFile) return direct;
  const withMd = app.vault.getAbstractFileByPath(
    clean.endsWith(".md") ? clean : `${clean}.md`
  );
  if (withMd instanceof TFile) return withMd;
  const dest = app.metadataCache.getFirstLinkpathDest(clean, "");
  return dest instanceof TFile ? dest : null;
}

async function openVaultFileAtLine(app: App | null, rawPath: string, lineNumberOneBased: number) {
  const targetApp = app ?? (typeof window !== "undefined" ? (window as any).app : null);
  if (!targetApp) return;

  const file = resolveVaultFile(targetApp, rawPath);
  if (!file) {
    new Notice(`Файл «${rawPath}» не найден.`);
    return;
  }

  const leaf = targetApp.workspace.getLeaf(false);
  await leaf.openFile(file);

  const lineIdx = Math.max(0, (lineNumberOneBased || 1) - 1);
  window.setTimeout(() => {
    try {
      const view: any = targetApp.workspace.activeLeaf?.view;
      const editor = view?.editor;
      if (editor) {
        editor.setCursor({ line: lineIdx, ch: 0 });
        editor.scrollIntoView(
          { from: { line: lineIdx, ch: 0 }, to: { line: lineIdx, ch: 0 } },
          true
        );
        editor.focus();
      }
    } catch {
      // Игнорируем
    }
  }, 120);
}

export async function loadBookAnnotationsFromVault(
  app: App,
  linkedBook: BookNoteInfo | null,
  wordsFilePath: string,
  quotesFilePath: string
): Promise<{ words: BookWordEntry[]; quotes: BookQuoteEntry[] }> {
  activeAnnotationsApp = app;
  activeWordsFilePath = wordsFilePath || "Words Cards";
  activeQuotesFilePath = quotesFilePath || "Знания/Цитаты из книг.md";
  configureTranslationCache(app);

  if (!linkedBook) return { words: [], quotes: [] };

  const words: BookWordEntry[] = [];
  const quotes: BookQuoteEntry[] = [];

  const wordsFile = resolveVaultFile(app, activeWordsFilePath);
  if (wordsFile) {
    try {
      const content = await app.vault.cachedRead(wordsFile);
      const lines = content.split(/\r?\n/);
      for (let i = 0; i < lines.length; i++) {
        const line = lines[i] ?? "";
        const srIdx = line.indexOf("<!--SR:");
        const cardPart = srIdx === -1 ? line : line.slice(0, srIdx);
        if (!/:{2,3}/.test(cardPart) || cardPart.trimStart().startsWith("#")) continue;

        const parts = cardPart.split(/:{2,3}/);
        const rawFront = (parts[0] ?? "").replace(/^[-*+]\s+/, "").trim();
        const rawBackJoined = parts.slice(1).join(" ").trim();

        let cleanBack = rawBackJoined;
        let wiki: string | null = null;

        const inlineMatch = rawBackJoined.match(
          /^(.*?)\s*\((?:источник|книга)::\s*(\[\[[^\]]+\]\])\)\s*(?![\s\S])/i
        );
        if (inlineMatch && inlineMatch[1] !== undefined && inlineMatch[2] !== undefined) {
          cleanBack = inlineMatch[1].trim();
          wiki = inlineMatch[2].trim();
        } else {
          const dotMatch = rawBackJoined.match(
            /^(.*?)\s*(?:·|\|)\s*(\[\[[^\]]+\]\])\s*(?![\s\S])/
          );
          if (dotMatch && dotMatch[1] !== undefined && dotMatch[2] !== undefined) {
            cleanBack = dotMatch[1].trim();
            wiki = dotMatch[2].trim();
          }
        }

        if (!wiki) continue;
        const target = extractLinkTarget(wiki);
        if (!isSameBookTarget(target, linkedBook)) continue;

        const exIdx = rawFront.indexOf(EXAMPLE_SEP);
        const word = exIdx === -1 ? rawFront : rawFront.slice(0, exIdx).trim();
        const example = exIdx === -1 ? "" : rawFront.slice(exIdx + EXAMPLE_SEP.length).trim();

        words.push({
          word,
          back: cleanBack,
          example,
          cfi: extractLinkCfi(wiki),
          lineNumber: i + 1,
        });
      }
    } catch {
      // Игнорируем ошибку чтения
    }
  }

  const quotesFile = resolveVaultFile(app, activeQuotesFilePath);
  if (quotesFile) {
    try {
      const content = await app.vault.cachedRead(quotesFile);
      const lines = content.split(/\r?\n/);
      for (let i = 0; i < lines.length; i++) {
        const line = (lines[i] ?? "").trim();
        if (!line.startsWith("-") || !line.includes("цитата::")) continue;

        const fields: Record<string, string> = {};
        const fieldRegex = /\(([а-яА-Яa-zA-Z0-9_]+)::\s*([^)]+)\)/g;
        for (const match of line.matchAll(fieldRegex)) {
          const k = match[1]?.trim();
          const v = match[2]?.trim();
          if (k && v) fields[k] = v;
        }

        const quoteText = fields["цитата"] ? decodeInline(fields["цитата"]) : "";
        if (!quoteText) continue;

        const linkTarget =
          extractLinkTarget(fields["книга_ссылка"]) || extractLinkTarget(fields["книга"]);
        const rawBookTitle = decodeInline(fields["книга"] || "");

        if (
          !isSameBookTarget(linkTarget, linkedBook) &&
          normStr(rawBookTitle) !== normStr(linkedBook.title)
        ) {
          continue;
        }

        const cfi = fields["позиция"] ? decodeInline(fields["позиция"]).trim() : null;
        quotes.push({
          text: quoteText,
          date: decodeInline(fields["дата"] || ""),
          cfi: cfi && cfi.startsWith("epubcfi(") ? cfi : null,
          lineNumber: i + 1,
        });
      }
    } catch {
      // Игнорируем
    }
  }

  return { words: words.reverse(), quotes: quotes.reverse() };
}

      text: opts.exampleOrDate,
    });
  }

  const actions = cardEl.createDiv("my-epub-hl-card-actions");

  if (opts.speakWord) {
    const wordToSpeak = opts.speakWord;
    const ttsBtn = actions.createEl("button", {
      cls: "my-epub-hl-card-btn",
      text: "🔊 Озвучить",
      attr: { type: "button", "aria-label": "Озвучить слово" },
    });
    ttsBtn.onclick = () => speakEnglishText(wordToSpeak);
  }

  const openLineBtn = actions.createEl("button", {
    cls: "mod-cta my-epub-hl-card-btn",
    text: `📄 Перейти в карточку (стр. ${opts.lineNumber})`,
    attr: { type: "button" },
  });
  openLineBtn.onclick = () => {
    closeActiveHighlightPopover();
    void openVaultFileAtLine(activeAnnotationsApp, opts.targetFilePath, opts.lineNumber);
  };

  const cleanupTargets: Array<() => void> = [];
  const onOutsideTap = (ev: Event) => {
    const target = ev.target as Node | null;
    if (target && cardEl.contains(target)) return;
    closeActiveHighlightPopover();
  };

  const timerId = window.setTimeout(() => {
    document.addEventListener("pointerdown", onOutsideTap, true);
    document.addEventListener("touchstart", onOutsideTap, { capture: true, passive: true });
    cleanupTargets.push(() => {
      document.removeEventListener("pointerdown", onOutsideTap, true);
      document.removeEventListener("touchstart", onOutsideTap, true);
    });

    try {
      const contentsList: any[] = (rendition as any)?.getContents?.() || [];
      for (const c of contentsList) {
        const doc: Document | undefined = c?.document;
        if (doc) {
          doc.addEventListener("pointerdown", onOutsideTap, true);
          doc.addEventListener("touchstart", onOutsideTap, { capture: true, passive: true });
          cleanupTargets.push(() => {
            doc.removeEventListener("pointerdown", onOutsideTap, true);
            doc.removeEventListener("touchstart", onOutsideTap, true);
          });
        }
      }
    } catch {
      // Игнорируем
    }
  }, 60);

  const onRelocatedClose = () => closeActiveHighlightPopover();
  try {
    rendition.on("relocated", onRelocatedClose);
  } catch {
    // Игнорируем
  }

  activeHighlightPopoverCleanup = () => {
    window.clearTimeout(timerId);
    for (const fn of cleanupTargets) {
      try {
        fn();
      } catch {
        // Игнорируем
      }
    }
    try {
      rendition.off?.("relocated", onRelocatedClose);
    } catch {
      // Игнорируем
    }
    cardEl.remove();
  };
}

export function applyHighlightsToRendition(
  rendition: Rendition | null,
  appliedCfis: Set<string>,
  words: BookWordEntry[],
  quotes: BookQuoteEntry[]
) {
  closeActiveHighlightPopover();
  if (!rendition || !rendition.annotations) return;

  for (const oldCfi of appliedCfis) {
    try {
      rendition.annotations.remove(oldCfi, "highlight");
    } catch {
      // Игнорируем
    }
  }
  appliedCfis.clear();

  for (const w of words) {
    if (!w.cfi || !w.cfi.startsWith("epubcfi(") || !w.cfi.includes(",")) continue;
    try {
      rendition.annotations.highlight(
        w.cfi,
        {},
        () => {
          speakEnglishText(w.word);
          showHighlightPopoverCard(rendition, {
            type: "word",
            title: w.word,
            translationOrQuote: w.back,
            exampleOrDate: w.example ? `«${w.example}»` : "",
            speakWord: w.word,
            targetFilePath: activeWordsFilePath,
            lineNumber: w.lineNumber,
          });
        },
        "my-epub-hl-word",
        highlightStyles("#599ed6", "0.28")
      );
      appliedCfis.add(w.cfi);
    } catch {
      // Игнорируем
    }
  }

  for (const q of quotes) {
    if (!q.cfi || !q.cfi.startsWith("epubcfi(") || !q.cfi.includes(",")) continue;
    try {
      rendition.annotations.highlight(
        q.cfi,
        {},
        () => {
          showHighlightPopoverCard(rendition, {
            type: "quote",
            title: "Цитата",
            translationOrQuote: `«${q.text}»`,
            exampleOrDate: q.date ? `Дата: ${q.date}` : "",
            targetFilePath: activeQuotesFilePath,
            lineNumber: q.lineNumber,
          });
        },
        "my-epub-hl-quote",
        highlightStyles("#e5a158", "0.30")
      );
      appliedCfis.add(q.cfi);
    } catch {
      // Игнорируем
    }
  }
}

    input.onkeydown = (e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        void runLookup();
      }
    };

    speakEnglishText(this.textToTranslate);
    await runLookup();
  }

  onClose() {
    this.contentEl.empty();
  }
}

// Вспомогательный ручной поиск по DOM секции, если item.find в epub.js недоступен
function manualSearchInSectionDom(
  sectionItem: any,
  query: string,
  maxNeeded: number
): Array<{ cfi: string; excerpt: string }> {
  const results: Array<{ cfi: string; excerpt: string }> = [];
  const doc: Document | undefined = sectionItem?.document;
  const body = doc?.body || doc?.documentElement;
  if (!doc || !body) return results;

  const qLower = query.toLowerCase();
  const walker = doc.createTreeWalker(body, NodeFilter.SHOW_TEXT);
  let node: Node | null = walker.nextNode();

  while (node && results.length < maxNeeded) {
    const text = node.nodeValue || "";
    const textLower = text.toLowerCase();
    let fromIdx = 0;

    while (results.length < maxNeeded) {
      const matchPos = textLower.indexOf(qLower, fromIdx);
      if (matchPos === -1) break;
      fromIdx = matchPos + Math.max(1, qLower.length);

      try {
        const range = doc.createRange();
        range.setStart(node, matchPos);
        range.setEnd(node, Math.min(text.length, matchPos + query.length));
        const cfi =
          typeof sectionItem.cfiFromRange === "function"
            ? sectionItem.cfiFromRange(range)
            : typeof sectionItem.cfiFromElement === "function" && node.parentElement
              ? sectionItem.cfiFromElement(node.parentElement)
              : "";

        if (cfi) {
          const startSnippet = Math.max(0, matchPos - 45);
          const endSnippet = Math.min(text.length, matchPos + query.length + 55);
          const prefix = startSnippet > 0 ? "…" : "";
          const suffix = endSnippet < text.length ? "…" : "";
          const excerpt = `${prefix}${text.slice(startSnippet, endSnippet).replace(/\s+/g, " ").trim()}${suffix}`;
          results.push({ cfi, excerpt });
        }
      } catch {
        // Игнорируем ошибку построения CFI для узла
      }
    }
    node = walker.nextNode();
  }

  return results;
}

// --- Единое модальное окно навигации: 📑 Главы | 🔖 Закладки | 🎓 Слова и цитаты | 🔍 Поиск ---
export class BookNavModal extends Modal {
  book: Book | null;
  toc: NavItem[];
  bookmarks: BookmarkItem[];
  words: BookWordEntry[];
  quotes: BookQuoteEntry[];
  currentChapterLabel: string;
  currentPage: number | null;
  currentPct: number | null;
  onSelectToc: (href: string) => void;
  onJumpCfi: (cfi: string) => void;
  onToggleBookmark: () => BookmarkItem[];
  onDeleteBookmark: (cfi: string) => BookmarkItem[];
  activeTab: "toc" | "bookmarks" | "items" | "search" = "toc";
  itemsFilter = "";
  searchQuery = "";
  searchResults: BookSearchMatch[] = [];
  searchStatusText = "";
  isSearching = false;
  searchAbortSeq = 0;
  currentTocMarked = false;
  currentTocEl: HTMLElement | null = null;

  constructor(
    app: App,
    opts: {
      book?: Book | null;
      toc: NavItem[];
      bookmarks: BookmarkItem[];
      words: BookWordEntry[];
      quotes: BookQuoteEntry[];
      currentChapterLabel: string;
      currentPage: number | null;
      currentPct: number | null;
      onSelectToc: (href: string) => void;
      onJumpCfi: (cfi: string) => void;
      onToggleBookmark: () => BookmarkItem[];
      onDeleteBookmark: (cfi: string) => BookmarkItem[];
    }
  ) {
    super(app);
    (this as any).shouldRestoreSelection = false;
    this.book = opts.book ?? null;
    this.toc = opts.toc;
    this.bookmarks = opts.bookmarks;
    this.words = opts.words;
    this.quotes = opts.quotes;
    this.currentChapterLabel = opts.currentChapterLabel;
    this.currentPage = opts.currentPage;
    this.currentPct = opts.currentPct;
    this.onSelectToc = opts.onSelectToc;
    this.onJumpCfi = opts.onJumpCfi;
    this.onToggleBookmark = opts.onToggleBookmark;
    this.onDeleteBookmark = opts.onDeleteBookmark;
  }

  safeClose() {
    this.searchAbortSeq++;
    this.isSearching = false;
    try {
      (this as any).shouldRestoreSelection = false;
      this.close();
    } catch {
      // Игнорируем
    }
  }

  onOpen() {
    (this as any).shouldRestoreSelection = false;
    this.render();
  }

  resolveChapterNameForHref(rawHref: string): string {
    const cleanHref = (rawHref || "").split("#")[0] || "";
    if (!cleanHref || !this.toc) return "";
    const findInItems = (items: NavItem[]): string => {
      for (const it of items) {
        const itemHref = (it.href || "").split("#")[0] || "";
        if (
          itemHref &&
          (itemHref === cleanHref || cleanHref.endsWith(itemHref) || itemHref.endsWith(cleanHref))
        ) {
          return (it.label || "").trim();
        }
        if (it.subitems && it.subitems.length > 0) {
          const sub = findInItems(it.subitems);
          if (sub) return sub;
        }
      }
      return "";
    };
    return findInItems(this.toc);
  }

  render() {
    const { contentEl } = this;
    contentEl.empty();
    contentEl.addClass("my-epub-toc-modal");

    const tabsRow = contentEl.createDiv("my-epub-nav-tabs");
    const totalItems = this.words.length + this.quotes.length;

    const tabs: Array<{ id: "toc" | "bookmarks" | "items" | "search"; label: string }> = [
      { id: "toc", label: "📑 Главы" },
      { id: "bookmarks", label: `🔖 Закладки (${this.bookmarks.length})` },
      { id: "items", label: `🎓 Слова (${totalItems})` },
      { id: "search", label: "🔍 Поиск" },
    ];

    for (const t of tabs) {
      const btn = tabsRow.createEl("button", {
        cls: "my-epub-nav-tab" + (this.activeTab === t.id ? " is-active" : ""),
        text: t.label,
      });
      btn.onclick = () => {
        if (this.activeTab === "search" && t.id !== "search") {
          this.searchAbortSeq++;
          this.isSearching = false;
        }
        this.activeTab = t.id;
        this.render();
      };
    }

    if (this.activeTab === "items") {
      this.renderItemsTab(contentEl);
      return;
    }

    if (this.activeTab === "search") {
      this.renderSearchTab(contentEl);
      return;
    }

    const body = contentEl.createDiv("my-epub-toc-list");

    if (this.activeTab === "toc") {
      if (!this.toc || this.toc.length === 0) {
        body.createDiv({ cls: "my-epub-toc-empty", text: "В этой книге нет оглавления." });
        return;
      }
      this.currentTocMarked = false;
      this.currentTocEl = null;
      this.renderTocItems(body, this.toc, 0);
      const target = this.currentTocEl as HTMLElement | null;
      if (target) {
        window.setTimeout(() => target.scrollIntoView({ block: "center" }), 0);
      }
      return;
    }

    // Закладки
    const addRow = body.createDiv("my-epub-bm-top");
    const pageStr = this.currentPage
      ? `стр. ${this.currentPage} (${this.currentPct ?? 0}%)`
      : "текущая страница";
    const addBmBtn = addRow.createEl("button", {
      cls: "mod-cta my-epub-modal-btn",
      text: `🔖 Сохранить закладку на ${pageStr}`,
    });
    addBmBtn.onclick = () => {
      this.bookmarks = this.onToggleBookmark();
      this.render();
    };

    if (this.bookmarks.length === 0) {
      body.createDiv({
        cls: "my-epub-toc-empty",
        text: "Закладок пока нет. Нажми кнопку выше, чтобы отметить текущую страницу.",
      });
      return;
    }

    const sorted = [...this.bookmarks].sort((a, b) => a.page - b.page);
    for (const bm of sorted) {
      const row = body.createDiv("my-epub-bm-item");
      const left = row.createDiv("my-epub-bm-main");
      left.createDiv({
        cls: "my-epub-bm-title",
        text: `🔖 Стр. ${bm.page} · ${bm.pct}%${bm.label ? " — " + bm.label : ""}`,
      });
      left.createDiv({ cls: "my-epub-bm-sub", text: bm.createdAt });
      left.onclick = () => {
        this.onJumpCfi(bm.cfi);
        this.safeClose();
      };

      const delBtn = row.createEl("button", {
        cls: "my-epub-bm-del",
        text: "✕",
        attr: { title: "Удалить закладку", "aria-label": "Удалить закладку" },
      });
      delBtn.onclick = (e) => {
        e.stopPropagation();
        this.bookmarks = this.onDeleteBookmark(bm.cfi);
        this.render();
      };
    }
  }

  // C1: Вкладка «🔍 Поиск» по главам книги через book.spine.each и item.find(query) (с фолбэком на item.load + DOM)
  renderSearchTab(parent: HTMLElement) {
    const controlsRow = parent.createDiv("my-epub-search-controls");
    const searchInput = controlsRow.createEl("input", {
      cls: "my-epub-nav-search my-epub-search-input",
      type: "search",
      value: this.searchQuery,
      placeholder: "Поиск фразы или слова по тексту книги…",
    });

    const runBtn = controlsRow.createEl("button", {
      cls: "mod-cta my-epub-modal-btn",
      text: "Найти",
    });

    const cancelBtn = controlsRow.createEl("button", {
      cls: "my-epub-modal-btn",
      text: "Стоп",
    });
    cancelBtn.style.display = this.isSearching ? "inline-flex" : "none";

    const statusEl = parent.createDiv({
      cls: "my-epub-search-status",
      text: this.searchStatusText,
    });
    statusEl.style.display = this.searchStatusText ? "block" : "none";

    const listEl = parent.createDiv("my-epub-toc-list");

    const drawResults = () => {
      listEl.empty();
      statusEl.setText(this.searchStatusText);
      statusEl.style.display = this.searchStatusText ? "block" : "none";
      cancelBtn.style.display = this.isSearching ? "inline-flex" : "none";
      runBtn.disabled = this.isSearching;

      if (!this.isSearching && this.searchResults.length === 0 && this.searchQuery.trim()) {
        listEl.createDiv({
          cls: "my-epub-toc-empty",
          text: this.searchStatusText ? "" : "Совпадений не найдено.",
        });
        return;
      }

      for (const item of this.searchResults) {
        const row = listEl.createDiv("my-epub-bm-item");
        const main = row.createDiv("my-epub-bm-main");
        if (item.chapterLabel) {
          main.createDiv({
            cls: "my-epub-search-chapter",
            text: `📑 ${item.chapterLabel}`,
          });
        }
        main.createDiv({
          cls: "my-epub-bm-title",
          text: item.excerpt,
        });
        main.onclick = () => {
          this.onJumpCfi(item.cfi);
          this.safeClose();
        };
      }
    };

    const startSearch = async () => {
      const q = searchInput.value.trim();
      this.searchQuery = q;
      if (q.length < 2) {
        this.searchStatusText = "Введи минимум 2 символа для поиска.";
        this.searchResults = [];
        drawResults();
        return;
      }
      if (!this.book || !this.book.spine) {
        this.searchStatusText = "Книга ещё не загружена.";
        drawResults();
        return;
      }

      const mySeq = ++this.searchAbortSeq;
      this.isSearching = true;
      this.searchResults = [];
      this.searchStatusText = "Подготовка к поиску…";
      drawResults();

      const spineItems: any[] = [];
      const spineAny: any = this.book.spine;
      if (typeof spineAny.each === "function") {
        spineAny.each((item: any) => spineItems.push(item));
      } else if (Array.isArray(spineAny.spineItems)) {
        spineItems.push(...spineAny.spineItems);
      }

      const totalSections = spineItems.length;

      for (let i = 0; i < totalSections; i++) {
        if (mySeq !== this.searchAbortSeq) return;
        const section = spineItems[i];
        if (!section) continue;

        this.searchStatusText = `Ищу в секции ${i + 1} из ${totalSections}… (найдено: ${this.searchResults.length})`;
        statusEl.setText(this.searchStatusText);
        statusEl.style.display = "block";

        // Уступаем поток UI между главами, чтобы кнопка «Стоп» оставалась отзывчивой
        await new Promise<void>((r) => window.setTimeout(r, 0));
        if (mySeq !== this.searchAbortSeq) return;

        let loadedHere = false;
        try {
          if (!section.document && typeof section.load === "function") {
            await section.load(this.book.load.bind(this.book));
            loadedHere = true;
          }

          if (mySeq !== this.searchAbortSeq) {
            if (loadedHere) section.unload?.();
            return;
          }

          const needed = MAX_SEARCH_RESULTS - this.searchResults.length;
          let sectionMatches: Array<{ cfi: string; excerpt: string }> = [];

          // Проверяем наличие item.find в используемой версии epub.js
          if (typeof section.find === "function") {
            try {
              const rawFound = section.find(q);
              if (Array.isArray(rawFound)) {
                sectionMatches = rawFound
                  .filter((m: any) => m && typeof m.cfi === "string" && m.cfi.length > 0)
                  .slice(0, needed)
                  .map((m: any) => ({
                    cfi: String(m.cfi),
                    excerpt: String(m.excerpt || "").replace(/\s+/g, " ").trim(),
                  }));
              }
            } catch {
              sectionMatches = manualSearchInSectionDom(section, q, needed);
            }
          } else {
            sectionMatches = manualSearchInSectionDom(section, q, needed);
          }

          if (sectionMatches.length > 0) {
            const chapterLabel =
              this.resolveChapterNameForHref(section.href || "") || `Раздел ${i + 1}`;
            for (const m of sectionMatches) {
              if (this.searchResults.length >= MAX_SEARCH_RESULTS) break;
              this.searchResults.push({
                cfi: m.cfi,
                excerpt: m.excerpt || q,
                chapterLabel,
              });
            }
            drawResults();
          }
        } catch {
          // Пропускаем секцию при ошибке чтения
        } finally {
          if (loadedHere) {
            try {
              section.unload?.();
            } catch {
              // Игнорируем
            }
          }
        }

        if (this.searchResults.length >= MAX_SEARCH_RESULTS) {
          break;
        }
      }

      if (mySeq !== this.searchAbortSeq) return;
      this.isSearching = false;
      const hitLimit = this.searchResults.length >= MAX_SEARCH_RESULTS;
      this.searchStatusText =
        this.searchResults.length > 0
          ? `Найдено совпадений: ${this.searchResults.length}${hitLimit ? ` (показаны первые ${MAX_SEARCH_RESULTS})` : ""}`
          : "Совпадений не найдено.";
      drawResults();
    };

    cancelBtn.onclick = () => {
      this.searchAbortSeq++;
      this.isSearching = false;
      this.searchStatusText = `Поиск остановлен (найдено: ${this.searchResults.length}).`;
      drawResults();
    };

    runBtn.onclick = () => void startSearch();
    searchInput.onkeydown = (e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        void startSearch();
      }
    };

export function formatDurationMinutes(minutesFloat: number): string {
  const mins = Math.max(1, Math.round(minutesFloat));
  if (mins < 60) return `~${mins} мин`;
  const h = Math.floor(mins / 60);
  const m = mins % 60;
  return m > 0 ? `~${h} ч ${m} мин` : `~${h} ч`;
}

export function accrueActiveReadingTime(view: EpubView, now = Date.now()) {
  if (!view.isInitialLoadDone || !view.lastReadingTickAt || !view.lastActivityAt) {
    view.lastReadingTickAt = now;
    return;
  }
  if (document.visibilityState === "hidden") {
    view.lastReadingTickAt = now;

  }

export function computeFastPercentage(
  view: EpubView,
  cfi: string,
  loc?: EpubRelocatedEvent
): number {
  if (!view.book) return view.approxProgressPct ?? 0;

  if (view.book.locations && view.book.locations.length() > 0) {
    const p = view.book.locations.percentageFromCfi(cfi);
    if (typeof p === "number" && Number.isFinite(p)) {
      return Math.max(0, Math.min(100, Math.round(p * 100)));
    }
  }

  const rendExt = view.rendition as unknown as EpubRenditionExtended | null;
  const start = loc?.start ?? rendExt?.currentLocation?.()?.start;
  const bookExt = view.book as unknown as EpubBookExtended;
  const spineLen = bookExt.spine?.spineItems?.length || 0;

  if (start && typeof start.index === "number" && spineLen > 0) {
    const dispPage = start.displayed?.page ?? 1;
    const dispTotal = Math.max(1, start.displayed?.total ?? 1);
    const frac = (start.index + (dispPage - 1) / dispTotal) / spineLen;
    const pct = Math.max(0, Math.min(100, Math.round(frac * 100)));
    view.approxProgressPct = pct;
    return pct;
  }

  return view.approxProgressPct ?? 0;
}

export function buildChapterSummaryText(view: EpubView): string {
  const parts: string[] = [];
  if (view.currentChapterTitle) parts.push(`Глава: «${view.currentChapterTitle}»`);

  const hasPaceData = view.turnDurationsMs.length >= 3;
  const avgTurnMs = hasPaceData
    ? view.turnDurationsMs.reduce((acc, v) => acc + v, 0) / view.turnDurationsMs.length
    : 0;

  if (view.chapterPageCurrent !== null && view.chapterPageTotal !== null) {
    const leftInChapter = Math.max(0, view.chapterPageTotal - view.chapterPageCurrent);
    if (leftInChapter === 0) {
      parts.push(`последняя страница главы (${view.chapterPageCurrent}/${view.chapterPageTotal})`);
    } else {
      const etaChapter =
        hasPaceData && avgTurnMs > 0
          ? `, ${formatDurationMinutes((leftInChapter * avgTurnMs) / 60_000)}`
          : "";
      parts.push(
        `до конца главы: ${leftInChapter} стр.${etaChapter} (${view.chapterPageCurrent} из ${view.chapterPageTotal})`
      );
    }
  }

  if (
    hasPaceData &&
    avgTurnMs > 0 &&
    view.currentPageNum !== null &&
    view.totalPagesNum !== null &&
    view.totalPagesNum > view.currentPageNum
  ) {
    const leftInBook = view.totalPagesNum - view.currentPageNum;
    parts.push(`до конца книги: ${formatDurationMinutes((leftInBook * avgTurnMs) / 60_000)}`);
  }

  return parts.join(" · ");
}

export function getCfiFromPageNumberSafe(view: EpubView, page: number): string | null {
  if (!view.book || !view.totalPagesNum || view.totalPagesNum <= 0) return null;

  const idx = clampedPage - 1;
  const locs = view.book.locations as unknown as EpubLocationsExtended;

  if (typeof locs.cfiFromLocation === "function") {
    try {
      const cfi = locs.cfiFromLocation(idx);
      if (typeof cfi === "string" && cfi.length > 0) return cfi;
    } catch {
      // Игнорируем
    }
  }

  const fromArr = locs._locations?.[idx];
  if (typeof fromArr === "string" && fromArr.length > 0) return fromArr;

  return view.book.locations.cfiFromPercentage(
    Math.max(0, Math.min(1, idx / Math.max(1, total)))

}

export async function flushBookYamlProgress(view: EpubView) {
  if (view.yamlSyncTimer) {
    window.clearTimeout(view.yamlSyncTimer);
    view.yamlSyncTimer = null;
  }
  if (view.navHistory.length > 0) return;

  accrueActiveReadingTime(view, Date.now());
  const addedReadingMinutes = Math.floor(view.unsavedReadingSeconds / 60);

  if (!view.pendingYamlSync && addedReadingMinutes <= 0) return;
  if (!view.linkedBookNote || !view.file) return;
  if (view.linkedBookNote.linkedEpubPath !== view.file.path) return;

  const noteFile = view.linkedBookNote.file;
  const pageNum = view.currentPageNum;
  const totalNum = view.totalPagesNum;
  const pctNum = view.currentProgressPct ?? view.approxProgressPct;
  const cfiToSync = view.stableCfi;
  const posTimeIso =
    view.lastLocalPositionSavedAt ||
    view.plugin.settings.progressUpdatedAt?.[view.file.path] ||
    new Date().toISOString();

  if (!cfiToSync) return;
  view.pendingYamlSync = false;

  const todayISO = getTodayISO();
  const fmCache = view.app.metadataCache.getFileCache(noteFile)?.frontmatter;
  if (
    addedReadingMinutes <= 0 &&
    fmCache &&
    (totalNum === null || fmCache["страниц"] === totalNum) &&
    (pageNum === null || fmCache["прочитано"] === pageNum) &&
    (pctNum === null || fmCache["прогресс"] === pctNum) &&
    fmCache["последнее_чтение"] === todayISO &&
    fmCache["позиция"] === cfiToSync &&
    fmCache["позиция_время"] === posTimeIso
  ) {
    return;
  }

  if (addedReadingMinutes > 0) {
    view.unsavedReadingSeconds -= addedReadingMinutes * 60;
  }

  try {
    await view.app.fileManager.processFrontMatter(noteFile, (fm: Record<string, unknown>) => {
      if (totalNum !== null && totalNum > 0 && fm["страниц"] !== totalNum) {
        fm["страниц"] = totalNum;
      }
      if (pageNum !== null && pageNum > 0 && fm["прочитано"] !== pageNum) {
        fm["прочитано"] = pageNum;
      }
      if (pctNum !== null && pctNum >= 0 && fm["прогресс"] !== pctNum) {
        fm["прогресс"] = pctNum;
      }
      if (fm["последнее_чтение"] !== todayISO) {
        fm["последнее_чтение"] = todayISO;
      }
      if (fm["позиция"] !== cfiToSync) {
        fm["позиция"] = cfiToSync;
      }
      if (fm["позиция_время"] !== posTimeIso) {
        fm["позиция_время"] = posTimeIso;
      }
      if (addedReadingMinutes > 0) {
        const prevMin = Number(fm["минут_чтения"]);
        const baseMin = Number.isFinite(prevMin) && prevMin >= 0 ? Math.round(prevMin) : 0;
        const nextMin = baseMin + addedReadingMinutes;
        if (fm["минут_чтения"] !== nextMin) {
          fm["минут_чтения"] = nextMin;
        }
      }
      if (!fm["статус"] || fm["статус"] === "В планах") {
        fm["статус"] = "Читаю";
      }
    });
  } catch (e) {
    if (addedReadingMinutes > 0) {
      view.unsavedReadingSeconds += addedReadingMinutes * 60;

// Модальное окно настроек вида и чтения вместо системного Menu
class AppearanceModal extends Modal {
  view: EpubView;

  constructor(app: App, view: EpubView) {
    super(app);
    (this as unknown as { shouldRestoreSelection?: boolean }).shouldRestoreSelection = false;
    this.view = view;
  }

  safeClose() {
    try {
      (this as unknown as { shouldRestoreSelection?: boolean }).shouldRestoreSelection = false;
      this.close();
    } catch {
      // Игнорируем
    }
  }

  onOpen() {
    (this as unknown as { shouldRestoreSelection?: boolean }).shouldRestoreSelection = false;
    this.render();
  }

  private renderChipGroup<T extends string | number>(
    parent: HTMLElement,
    label: string,
    options: Array<{ id: T; label: string }>,
    isActive: (id: T) => boolean,
    onPick: (id: T, chosenLabel: string) => void
  ) {
    const sec = parent.createDiv("my-epub-appear-section");
    sec.createDiv({ cls: "my-epub-appear-label", text: label });
    const row = sec.createDiv("my-epub-appear-chips");
    for (const opt of options) {
      const btn = row.createEl("button", {
        cls: "my-epub-appear-chip" + (isActive(opt.id) ? " is-active" : ""),
        text: opt.label,
        attr: { type: "button" },
      });
      btn.onclick = (e) => {
        e.preventDefault();
        e.stopPropagation();
        onPick(opt.id, opt.label);
        this.render();
      };
    }
  }

  render() {
    const { contentEl } = this;
    const prevScrollTop = contentEl.scrollTop;
    contentEl.empty();
    contentEl.addClass("my-epub-modal", "my-epub-appear-modal");

    const ver = this.view.plugin.manifest?.version || "0.2.8";
    const head = contentEl.createDiv("my-epub-word-head");
    head.createSpan({
      cls: "my-epub-word-title",
      text: `🔤 Вид и настройки чтения (${ver})`,
    });

    const s = this.view.plugin.settings;
    const curTheme = s.themeMode ?? "auto";
    const curSpread = s.spreadMode ?? "none";
    const curFont = s.fontFamily ?? "serif";
    const curLh = s.lineHeight ?? 1.6;
    const curMargin = s.pageMarginPx ?? 12;
    const curAlign = s.textAlign ?? "default";
    const curHyphens = Boolean(s.hyphens);
    const curAnim = s.pageAnimation ?? "slide";
    const curPos = s.barPosition ?? "bottom";
    const curWidth = s.desktopWidthPct ?? 85;

    const topActions = contentEl.createDiv("my-epub-appear-actions");

    const linkedLabel = this.view.linkedBookNote
      ? `📖 Книга: ${this.view.linkedBookNote.title} (сменить…)`
      : "📖 Привязать заметку книги…";
    const linkBookBtn = topActions.createEl("button", {
      cls: "my-epub-appear-action-btn",
      text: linkedLabel,
      attr: { type: "button" },
    });
    linkBookBtn.onclick = () => {
      this.safeClose();
      this.view.openBookLinkModal();
    };

    const isBm = this.view.isCurrentPageBookmarked();
    const bmBtn = topActions.createEl("button", {
      cls: "my-epub-appear-action-btn" + (isBm ? " is-active" : ""),
      text: isBm
        ? "🔖 Удалить закладку со страницы"
        : "🔖 Добавить закладку на странице",
      attr: { type: "button" },
    });
    bmBtn.onclick = () => {
      this.view.toggleBookmarkOnCurrentPage();
      this.render();
    };

    if (this.view.totalPagesNum && this.view.totalPagesNum > 0) {
      const jumpBtn = topActions.createEl("button", {
        cls: "my-epub-appear-action-btn",
        text: `🧭 Перейти к странице (сейчас ${this.view.currentPageNum ?? 1} из ${this.view.totalPagesNum})…`,
        attr: { type: "button" },
      });
      jumpBtn.onclick = () => {
        this.safeClose();
        this.view.openJumpModal();
      };
    }

    const fontSec = contentEl.createDiv("my-epub-appear-section");
    fontSec.createDiv({
      cls: "my-epub-appear-label",
      text: `Размер шрифта: ${s.fontSize}%`,
    });
    const fontRow = fontSec.createDiv("my-epub-appear-chips");
    const decFontBtn = fontRow.createEl("button", {
      cls: "my-epub-appear-chip",
      text: "A− Уменьшить",
      attr: { type: "button" },
    });
    decFontBtn.onclick = () => {
      this.view.setFont(-10);
      this.render();
    };
    const resetFontBtn = fontRow.createEl("button", {
      cls: "my-epub-appear-chip" + (s.fontSize === 120 ? " is-active" : ""),
      text: "120% (Сброс)",
      attr: { type: "button" },
    });
    resetFontBtn.onclick = () => {
      s.fontSize = 120;
      this.view.applyStylesAndRestore();
      this.view.plugin.requestSaveSettings();
      new Notice("Размер шрифта: 120%");
      this.render();
    };
    const incFontBtn = fontRow.createEl("button", {
      cls: "my-epub-appear-chip",
      text: "A+ Увеличить",
      attr: { type: "button" },
    });
    incFontBtn.onclick = () => {
      this.view.setFont(10);
      this.render();
    };

    this.renderChipGroup<ThemeMode>(
      contentEl,
      "Цветовая тема",
      [
        { id: "auto", label: "🌗 Авто (Obsidian)" },
        { id: "sepia", label: "📜 Сепия" },
        { id: "night", label: "🌙 Ночь" },
      ],
      (id) => curTheme === id,
      (id, label) => {
        s.themeMode = id;
        this.view.updateThemeBtnText();
        this.view.applyStyles();
        this.view.plugin.requestSaveSettings();
        new Notice(`Тема: ${label}`);
      }
    );

    this.renderChipGroup<SpreadMode>(
      contentEl,
      "Книжный разворот",
      [
        { id: "none", label: "📄 1 страница" },
        { id: "auto", label: "📖 2 страницы (на широком экране)" },
      ],
      (id) => curSpread === id,
      (id, label) => {
        this.view.applySpreadMode(id);
        this.view.plugin.requestSaveSettings();
        new Notice(`Разворот: ${label}`);
      }
    );

    this.renderChipGroup<FontFamily>(
      contentEl,
      "Шрифт",
      [
        { id: "serif", label: "Georgia (С засечками)" },
        { id: "sans", label: "Без засечек" },
        { id: "default", label: "Из книги" },
        { id: "mono", label: "Моноширинный" },
      ],
      (id) => curFont === id,
      (id, label) => {
        s.fontFamily = id;
        this.view.applyStylesAndRestore();
        this.view.plugin.requestSaveSettings();
        new Notice(`Шрифт: ${label}`);
      }
    );

    this.renderChipGroup<number>(
      contentEl,
      "Межстрочный интервал",
      [
        { id: 1.35, label: "Плотный (1.35)" },
        { id: 1.6, label: "Комфортный (1.6)" },
        { id: 1.85, label: "Свободный (1.85)" },
      ],
      (id) => Math.abs(curLh - id) < 0.05,
      (id, label) => {
        s.lineHeight = id;
        this.view.applyStylesAndRestore();
        this.view.plugin.requestSaveSettings();
        new Notice(`Интервал: ${label}`);
      }
    );

    this.renderChipGroup<number>(
      contentEl,
      "Боковые поля страницы",
      [
        { id: 6, label: "Узкие (6 px)" },
        { id: 12, label: "Стандартные (12 px)" },
        { id: 28, label: "Широкие (28 px)" },
      ],
      (id) => Math.abs(curMargin - id) <= 2,
      (id, label) => {
        s.pageMarginPx = id;
        this.view.applyStylesAndRestore();
        this.view.plugin.requestSaveSettings();
        new Notice(`Поля: ${label}`);
      }
    );

    const alignSec = contentEl.createDiv("my-epub-appear-section");
    alignSec.createDiv({ cls: "my-epub-appear-label", text: "Выравнивание и переносы" });
    const alignRow = alignSec.createDiv("my-epub-appear-chips");

    const aligns: Array<{ id: TextAlignMode; label: string }> = [
      { id: "default", label: "По умолчанию" },
      { id: "justify", label: "По ширине" },
      { id: "left", label: "По левому краю" },
    ];
    for (const al of aligns) {
      const btn = alignRow.createEl("button", {
        cls: "my-epub-appear-chip" + (curAlign === al.id ? " is-active" : ""),
        text: al.label,
        attr: { type: "button" },
      });
      btn.onclick = (e) => {
        e.preventDefault();
        e.stopPropagation();
        s.textAlign = al.id;
        this.view.applyStylesAndRestore();
        this.view.plugin.requestSaveSettings();
        new Notice(`Выравнивание: ${al.label}`);
        this.render();
      };
    }

    const hyphBtn = alignRow.createEl("button", {
      cls: "my-epub-appear-chip" + (curHyphens ? " is-active" : ""),
      text: curHyphens ? "✓ Переносы слов" : "Переносы слов",
      attr: { type: "button" },
    });
    hyphBtn.onclick = (e) => {
      e.preventDefault();
      e.stopPropagation();
      s.hyphens = !curHyphens;
      this.view.applyStylesAndRestore();
      this.view.plugin.requestSaveSettings();
      new Notice(s.hyphens ? "Переносы слов включены" : "Переносы слов выключены");
      this.render();
    };

    if (Platform.isDesktop) {
      this.renderChipGroup<number>(
        contentEl,
        "Ширина текста (ПК)",
        [
          { id: 65, label: "65%" },
          { id: 75, label: "75%" },
          { id: 85, label: "85%" },
          { id: 100, label: "100%" },
        ],
        (id) => Math.abs(curWidth - id) < 3,
        (id, label) => {
          s.desktopWidthPct = id;
          this.view.applyDesktopWidth(true);
          this.view.plugin.requestSaveSettings();
          new Notice(`Ширина (ПК): ${label}`);
        }
      );
    }

    this.renderChipGroup<PageAnimation>(
      contentEl,
      "Анимация перелистывания",
      [
        { id: "slide", label: "Слайд" },
        { id: "fade", label: "Мягкое затухание" },
        { id: "none", label: "Выключена" },
      ],
      (id) => curAnim === id,
      (id, label) => {
        s.pageAnimation = id;
        this.view.plugin.requestSaveSettings();
        new Notice(`Анимация: ${label}`);
      }
    );

    this.renderChipGroup<BarPosition>(
      contentEl,
      "Положение панели кнопок",
      [
        { id: "bottom", label: "Снизу" },
        { id: "top", label: "Сверху" },
      ],
      (id) => curPos === id,
      (id, label) => {
        s.barPosition = id;
        this.view.isResizing = true;
        this.view.applyBarPosition();
        this.view.plugin.requestSaveSettings();
        this.view.triggerResize(true);
        new Notice(`Панель: ${label}`);
      }
    );

    contentEl.scrollTop = prevScrollTop;
  }

  onClose() {
    this.contentEl.empty();
  }
}

export class EpubView extends FileView {
  plugin: MyEpubPlugin;
  book: Book | null = null;
  rendition: Rendition | null = null;
  wrapEl!: HTMLElement;
  boxEl!: HTMLElement;
  barEl!: HTMLElement;
  fsBarEl!: HTMLElement;
  fsInfoEl!: HTMLElement;
  backPillEl!: HTMLElement;
  backPillTextEl!: HTMLElement;
  info!: HTMLElement;
  themeBtn!: HTMLButtonElement;
  fsBtn!: HTMLButtonElement;
  quickTrBtn!: HTMLButtonElement;
  addWordBtn!: HTMLButtonElement;
  addQuoteBtn!: HTMLButtonElement;
  fsQuickTrBtn!: HTMLButtonElement;
  fsAddWordBtn!: HTMLButtonElement;
  fsAddQuoteBtn!: HTMLButtonElement;
  resizeObserver: ResizeObserver | null = null;
  resizeTimer: number | null = null;
  animTimer: number | null = null;
  yamlSyncTimer: number | null = null;
  readingTickTimer: number | null = null;
  isFullscreen = false;
  lastGestureTime = 0;

  private workspaceEventsRegistered = false;
  private selectionTimers = new Set<number>();

  turnDurationsMs: number[] = [];
  lastTransitionTimestamp = 0;

  lastActivityAt = 0;
  lastReadingTickAt = 0;
  unsavedReadingSeconds = 0;

  stableCfi: string | null = null;
  lastLocalPositionSavedAt: string | null = null;
  isResizing = false;
  isInitialLoadDone = false;
  isGeneratingLocations = false;
  ExplicitCfiJumpPending = false;
  lastBoxWidth = 0;
  lastBoxHeight = 0;
  loadId = 0;
  cleanupListeners: Array<() => void> = [];

  navHistory: NavHistoryItem[] = [];

  capturedWord = "";
  capturedQuote = "";
  capturedCfi: string | null = null;
  extractedSentence = "";
  lastSelectionAt = 0;
  linkedBookNote: BookNoteInfo | null = null;
  bookAuthorMeta = "";
  bookTitleMeta = "";

  bookWords: BookWordEntry[] = [];
  bookQuotes: BookQuoteEntry[] = [];
  appliedHighlightCfis = new Set<string>();

  currentPageNum: number | null = null;
  totalPagesNum: number | null = null;
  currentProgressPct: number | null = null;
  approxProgressPct: number | null = null;
  chapterPageCurrent: number | null = null;
  chapterPageTotal: number | null = null;
  currentChapterTitle = "";
  pendingYamlSync = false;

  constructor(leaf: WorkspaceLeaf, plugin: MyEpubPlugin) {
    super(leaf);
    this.plugin = plugin;
  }

  getViewType() { return VIEW_TYPE; }
  getDisplayText() { return this.file?.basename ?? "EPUB"; }
  canAcceptExtension(ext: string) { return ext === "epub"; }

  async onOpen() {
    await super.onOpen();
    this.ensureWorkspaceEventsRegistered();
  }

  private ensureWorkspaceEventsRegistered() {
    if (this.workspaceEventsRegistered) return;
    this.workspaceEventsRegistered = true;
    this.registerEvent(
      this.app.workspace.on("css-change", () => this.applyStyles())
    );
    this.registerEvent(
      this.app.workspace.on("active-leaf-change", () => {
        this.plugin.syncGlobalBodyClasses();
        this.recordReadingActivity();
      })
    );
  }

  recordReadingActivity() {
    const now = Date.now();
    accrueActiveReadingTime(this, now);
    this.lastActivityAt = now;
    this.lastReadingTickAt = now;
  }

  private startReadingTimer() {
    if (this.readingTickTimer !== null) window.clearInterval(this.readingTickTimer);
    const now = Date.now();
    this.lastActivityAt = now;
    this.lastReadingTickAt = now;
    this.readingTickTimer = window.setInterval(() => {
      accrueActiveReadingTime(this, Date.now());
      if (this.unsavedReadingSeconds >= 60) {
        this.requestYamlProgressSync();
      }
    }, READING_TICK_INTERVAL_MS);
  }

  scheduleSelectionTimer(fn: () => void, ms: number): number {
    const id = window.setTimeout(() => {
      this.selectionTimers.delete(id);
      fn();
    }, ms);
    this.selectionTimers.add(id);
    return id;
  }

  clearSelectionTimer(id: number) {
    window.clearTimeout(id);
    this.selectionTimers.delete(id);
  }

  async onLoadFile(file: TFile) {
    this.ensureWorkspaceEventsRegistered();
    await this.flushYamlProgress();
    this.cleanupView();
    const currentLoadId = ++this.loadId;

    await this.plugin.reloadSettingsFromDisk();
    if (currentLoadId !== this.loadId) return;

    const root = this.contentEl;
    root.empty();
    root.addClass("my-epub");
    root.setAttr("tabindex", "-1");

    this.isResizing = false;
    this.isInitialLoadDone = false;
    this.isGeneratingLocations = false;
    this.lastBoxWidth = 0;
    this.lastBoxHeight = 0;
    this.navHistory = [];
    this.turnDurationsMs = [];
    this.lastTransitionTimestamp = Date.now();
    this.unsavedReadingSeconds = 0;
    this.capturedWord = "";
    this.capturedQuote = "";
    this.capturedCfi = null;
    this.extractedSentence = "";
    this.lastSelectionAt = 0;
    this.linkedBookNote = null;
    this.bookTitleMeta = file.basename;
    this.bookAuthorMeta = "";
    this.bookWords = [];
    this.bookQuotes = [];
    this.appliedHighlightCfis.clear();
    this.currentPageNum = null;
    this.totalPagesNum = null;
    this.currentProgressPct = null;
    this.approxProgressPct = null;
    this.chapterPageCurrent = null;
    this.chapterPageTotal = null;
    this.currentChapterTitle = "";
    this.pendingYamlSync = false;

    this.trySilentResolveLinkedBookNote(file);
    const initialPos = resolveInitialCfiBeforeRender(this, file);
    this.stableCfi = initialPos.cfi;
    this.lastLocalPositionSavedAt = initialPos.savedAt;

    this.wrapEl = root.createDiv("my-epub-wrap");
    this.boxEl = this.wrapEl.createDiv("my-epub-box");
    this.applyDesktopWidth(false);
    this.applySpreadClass();

    this.backPillEl = root.createDiv("my-epub-back-pill");
    this.backPillTextEl = this.backPillEl.createSpan({
      cls: "my-epub-back-pill-text",
      text: "↩ Назад",
    });
    const backDismissBtn = this.backPillEl.createSpan({
      cls: "my-epub-back-pill-close",
      text: "✕",
      attr: {
        title: "Скрыть кнопку возврата и остаться здесь",
        "aria-label": "Скрыть кнопку возврата",
      },
    });
    this.backPillTextEl.onclick = (e) => {
      e.stopPropagation();
      void this.popNavHistory();
    };
    backDismissBtn.onclick = (e) => {
      e.stopPropagation();
      this.navHistory = [];
      this.updateBackPillState();
      this.requestYamlProgressSync();
    };
    this.updateBackPillState();

    this.fsBarEl = root.createDiv("my-epub-fs-bar");
    const fsPrev = this.fsBarEl.createEl("button", {
      cls: "my-epub-fs-btn",
      text: "◀",
      attr: { "aria-label": "Предыдущая страница" },
    });
    this.fsInfoEl = this.fsBarEl.createSpan({
      cls: "my-epub-fs-info",
      text: "… · ⛶",
      attr: { "aria-label": "Выйти из полноэкранного режима" },
    });
    this.fsQuickTrBtn = this.fsBarEl.createEl("button", {
      cls: "my-epub-fs-btn my-epub-word-btn",
      text: "🌐",
      attr: { "aria-label": "Быстрый перевод" },
    });
    this.fsAddWordBtn = this.fsBarEl.createEl("button", {
      cls: "my-epub-fs-btn my-epub-word-btn",
      text: "+ Слово",
      attr: { "aria-label": "Добавить слово в словарь" },
    });
    this.fsAddQuoteBtn = this.fsBarEl.createEl("button", {
      cls: "my-epub-fs-btn my-epub-word-btn",
      text: "💬",
      attr: { "aria-label": "Добавить цитату" },
    });
    const fsNext = this.fsBarEl.createEl("button", {
      cls: "my-epub-fs-btn",
      text: "▶",
      attr: { "aria-label": "Следующая страница" },
    });

    fsPrev.onclick = (e) => { e.stopPropagation(); this.turnPage("prev"); };
    fsNext.onclick = (e) => { e.stopPropagation(); this.turnPage("next"); };
    this.fsInfoEl.onclick = (e) => { e.stopPropagation(); this.toggleFullscreen(false); };
    bindActionButton(this, this.fsQuickTrBtn, () => this.handleQuickTranslateClick());
    bindActionButton(this, this.fsAddWordBtn, () => this.handleAddWordClick());
    bindActionButton(this, this.fsAddQuoteBtn, () => this.handleAddQuoteClick());

    this.barEl = root.createDiv("my-epub-bar");
    this.applyBarPosition();

    const leftGroup = this.barEl.createDiv("my-epub-group");
    const prev = leftGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "◀",
      attr: { "aria-label": "Предыдущая страница" },
    });
    const tocBtn = leftGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "📑",
      attr: { "aria-label": "Оглавление, закладки, слова и поиск" },
    });
    const smaller = leftGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "A−",
      attr: { "aria-label": "Уменьшить размер шрифта" },
    });

    this.info = this.barEl.createDiv({
      cls: "my-epub-info",
      text: "загрузка…",
      attr: { title: "Нажми для быстрого перехода к странице" },
    });
    this.info.onclick = () => this.openJumpModal();

    const rightGroup = this.barEl.createDiv("my-epub-group");
    this.quickTrBtn = rightGroup.createEl("button", {
      cls: "my-epub-btn my-epub-word-btn",
      text: "🌐",
      attr: { "aria-label": "Быстрый перевод без записи в словарь" },
    });
    this.addWordBtn = rightGroup.createEl("button", {
      cls: "my-epub-btn my-epub-word-btn",
      text: "+ Слово",
      attr: { "aria-label": "Добавить выделенное слово в словарь" },
    });
    this.addQuoteBtn = rightGroup.createEl("button", {
      cls: "my-epub-btn my-epub-word-btn",
      text: "💬",
      attr: { "aria-label": "Добавить выделенную цитату" },
    });
    bindActionButton(this, this.quickTrBtn, () => this.handleQuickTranslateClick());
    bindActionButton(this, this.addWordBtn, () => this.handleAddWordClick());
    bindActionButton(this, this.addQuoteBtn, () => this.handleAddQuoteClick());
    this.updateSelectionButtonsState();

    const bigger = rightGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "A+",
      attr: { "aria-label": "Увеличить размер шрифта" },
    });
    const fontBtn = rightGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "🔤",
      attr: { "aria-label": "Вид, шрифт, поля, выравнивание и привязка книги" },
    });
    this.themeBtn = rightGroup.createEl("button", { cls: "my-epub-btn" });
    this.updateThemeBtnText();
    this.fsBtn = rightGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "⛶",
      attr: { "aria-label": "Полноэкранный режим" },
    });
    const next = rightGroup.createEl("button", {
      cls: "my-epub-btn",
      text: "▶",
      attr: { "aria-label": "Следующая страница" },
    });

    try {
      const data = await this.app.vault.readBinary(file);
      if (currentLoadId !== this.loadId) return;

      if (!data || data.byteLength === 0) {
        throw new Error("Файл книги пуст (0 байт).");
      }

      this.book = ePub(data);
      const spreadMode = this.plugin.settings.spreadMode ?? "none";
      this.rendition = this.book.renderTo(this.boxEl, {
        width: "100%",
        height: "100%",
        flow: "paginated",
        spread: spreadMode,
      });

      this.applyStyles();
      setupViewGesturesAndSelection(this);

      prev.onclick = () => this.turnPage("prev");
      next.onclick = () => this.turnPage("next");
      smaller.onclick = () => this.setFont(-10);
      bigger.onclick = () => this.setFont(10);
      this.themeBtn.onclick = () => this.toggleThemeMode();
      this.fsBtn.onclick = () => this.toggleFullscreen();
      tocBtn.onclick = () => this.openToc();
      fontBtn.onclick = () => this.openTypographyMenu();

      this.rendition.on("relocated", (loc: EpubRelocatedEvent) =>
        this.onRelocated(file, loc)
      );

      try {
        await this.rendition.display(this.stableCfi || undefined);
      } catch {
        this.stableCfi = null;
        await this.rendition.display();
      }

      if (currentLoadId !== this.loadId) return;

      this.lastBoxWidth = this.boxEl.clientWidth;
      this.lastBoxHeight = this.boxEl.clientHeight;
      this.setupResizeObserver();
      this.startReadingTimer();

      await this.book.ready;
      if (currentLoadId !== this.loadId || !this.book) return;

      try {
        const bookExt = this.book as unknown as EpubBookExtended;
        const meta = await bookExt.loaded?.metadata;
        if (meta?.title) this.bookTitleMeta = String(meta.title).trim();
        if (meta?.creator) this.bookAuthorMeta = String(meta.creator).trim();
      } catch {
        // Игнорируем
      }

      this.trySilentResolveLinkedBookNote(file);
      await this.refreshBookHighlights();
      if (currentLoadId !== this.loadId || !this.book || !this.rendition) return;

      const loadedFromCache = await tryLoadCachedLocations(this, file, currentLoadId);
      if (currentLoadId !== this.loadId || !this.book || !this.rendition) return;

      const rendExt = this.rendition as unknown as EpubRenditionExtended;
      const cur = rendExt.currentLocation?.();
      if (cur?.start?.cfi) {
        captureChapterMetricsFromLoc(this, cur);
        this.stableCfi = cur.start.cfi;
        this.updateProgressLabels(cur.start.cfi);
      }

      if (loadedFromCache) {
        await reconcilePositionWithBookNote(this, file, currentLoadId);
        if (currentLoadId !== this.loadId) return;
        this.isInitialLoadDone = true;
        this.ExplicitCfiJumpPending = false;
        this.requestYamlProgressSync();
      } else {
        this.isGeneratingLocations = true;
        if (this.stableCfi) this.updateProgressLabels(this.stableCfi);
        void generateLocationsInBackground(this, file, currentLoadId);
      }
    } catch (err: unknown) {
      if (currentLoadId !== this.loadId) return;
      console.error("MyEpubPlugin error loading file:", err);
      this.showError(file.name, getErrorMessage(err));
    }
  }

  pushNavHistory() {
    if (!this.stableCfi || !this.currentPageNum) return;
    const last = this.navHistory[this.navHistory.length - 1];
    if (last && last.page === this.currentPageNum) return;
    this.navHistory.push({
      cfi: this.stableCfi,
      page: this.currentPageNum,
    });
    if (this.navHistory.length > 10) this.navHistory.shift();
    this.updateBackPillState();
  }

  async popNavHistory() {
    const prev = this.navHistory.pop();
    this.updateBackPillState();
    if (!prev) return;
    await this.jumpToCfi(prev.cfi, false);
  }

  updateBackPillState() {
    if (!this.backPillEl || !this.backPillTextEl) return;
    const last = this.navHistory[this.navHistory.length - 1];
    if (!last) {
      this.backPillEl.removeClass("is-visible");
      return;
    }
    this.backPillTextEl.setText(`↩ Назад к стр. ${last.page}`);
    this.backPillEl.addClass("is-visible");
  }

  toggleBookmarkOnCurrentPage(): BookmarkItem[] {
    if (!this.file || !this.stableCfi) return [];
    if (!this.plugin.settings.bookmarks) this.plugin.settings.bookmarks = {};
    const list = this.plugin.settings.bookmarks[this.file.path] ?? [];
    const page = this.currentPageNum ?? 1;
    const pct = this.currentProgressPct ?? this.approxProgressPct ?? 0;

    const existingIdx = list.findIndex((b) => b.page === page || b.cfi === this.stableCfi);
    if (existingIdx !== -1) {
      list.splice(existingIdx, 1);
      new Notice(`Закладка на стр. ${page} удалена`);
    } else {
      list.unshift({
        cfi: this.stableCfi,
        page,
        pct,
        label: this.currentChapterTitle || this.bookTitleMeta || this.file.basename,
        createdAt: getTodayISO(),
      });
      new Notice(`🔖 Добавлена закладка: стр. ${page} (${pct}%)`);
    }

    this.plugin.settings.bookmarks[this.file.path] = list;
    this.plugin.requestSaveSettings();
    this.updateProgressLabels(this.stableCfi);
    return list;
  }

  deleteBookmarkByCfi(cfi: string): BookmarkItem[] {
    if (!this.file || !this.plugin.settings.bookmarks) return [];
    const list = (this.plugin.settings.bookmarks[this.file.path] ?? []).filter(
      (b) => b.cfi !== cfi
    );
    this.plugin.settings.bookmarks[this.file.path] = list;
    this.plugin.requestSaveSettings();
    if (this.stableCfi) this.updateProgressLabels(this.stableCfi);
    return list;
  }

  prepareMainWindowFocusBeforeModal() {

      for (const f of iframes) {
        f.contentWindow?.getSelection?.()?.removeAllRanges();
        (f.contentDocument?.activeElement as HTMLElement | null)?.blur?.();
      }
      window.getSelection?.()?.removeAllRanges();
      this.contentEl?.focus?.({ preventScroll: true });
    } catch {
      // Игнорируем
    }
  }

  openJumpModal() {
    if (!this.book) return;
    if (!this.totalPagesNum || this.totalPagesNum <= 0) {
      new Notice("Страницы ещё считаются — подожди пару секунд.");
      return;
    }
    this.prepareMainWindowFocusBeforeModal();
    const curPage = this.currentPageNum ?? 1;
    new JumpToPageModal(
      this.app,
      curPage,
      this.totalPagesNum,
      this.getChapterSummaryText(),
      (targetPage) => {
        const targetCfi = getCfiFromPageNumberSafe(this, targetPage);
        if (targetCfi) {
          this.pushNavHistory();
          void this.jumpToCfi(targetCfi, false);
        }
      }
    ).open();
  }

  async jumpToCfi(cfi: string, recordHistory = false) {
    if (!cfi) return;
    this.recordReadingActivity();
    if (recordHistory) this.pushNavHistory();
    const navCfi = cfi.includes(",") ? cfi.replace(/,[^,)]+\)$/, ")") : cfi;
    const nowIso = new Date().toISOString();
    this.ExplicitCfiJumpPending = true;
    this.stableCfi = navCfi;
    this.lastLocalPositionSavedAt = nowIso;
    if (this.file) {
      this.plugin.setLocalBookProgress(this.file.path, navCfi, nowIso);
    }
    if (this.rendition) {
      this.isResizing = false;
      try {
        await this.rendition.display(navCfi);
      } catch {
        // Игнорируем
      }
      this.updateProgressLabels(navCfi);
      if (this.isInitialLoadDone) {
        this.requestYamlProgressSync();
      }
    }
    this.ExplicitCfiJumpPending = false;
  }

  clearCapturedSelection() {
    this.capturedWord = "";
    this.capturedQuote = "";
    this.capturedCfi = null;
    this.extractedSentence = "";
    this.updateSelectionButtonsState();
    this.prepareMainWindowFocusBeforeModal();
  }

  trySilentResolveLinkedBookNote(epubFile: TFile) {
    const folder = this.plugin.settings.booksFolder || "Знания/Книги";
    const { exactByProp } = findMatchingBookNotes(this.app, epubFile, folder);
    if (exactByProp.length === 1 && exactByProp[0]) {
      this.linkedBookNote = exactByProp[0];
    }
  }

  openBookLinkModal(onAfterLinked?: (book: BookNoteInfo) => void) {
    if (!this.file) return;
    this.prepareMainWindowFocusBeforeModal();
    const folder = this.plugin.settings.booksFolder || "Знания/Книги";
    const { exactByProp, byTitle, all } = findMatchingBookNotes(this.app, this.file, folder);
    const candidates = exactByProp.length > 0 ? exactByProp : byTitle;

    new BookLinkModal(
      this.app,
      this.file,
      folder,
      candidates,
      all,
      this.bookTitleMeta || this.file.basename,
      this.bookAuthorMeta || "",
      this.currentPageNum,
      this.totalPagesNum,
      this.currentProgressPct,
      (resolvedBook) => {
        this.linkedBookNote = resolvedBook;
        this.requestYamlProgressSync();
        void this.refreshBookHighlights();
        if (onAfterLinked) onAfterLinked(resolvedBook);
      }
    ).open();
  }

  requestYamlProgressSync() {
    if (!this.isInitialLoadDone) return;
    if (this.navHistory.length > 0) return;
    if (!this.linkedBookNote || !this.file) return;
    if (this.linkedBookNote.linkedEpubPath !== this.file.path) return;
    if (!this.stableCfi) return;

    this.pendingYamlSync = true;
    if (this.yamlSyncTimer) window.clearTimeout(this.yamlSyncTimer);
    this.yamlSyncTimer = window.setTimeout(() => {
      this.yamlSyncTimer = null;
      void this.flushYamlProgress();
    }, YAML_SYNC_DEBOUNCE_MS);
  }

  async flushYamlProgress() {
    await flushBookYamlProgress(this);
  }

  updateSelectionButtonsState() {
    const hasWord = this.capturedWord.length > 0;
    const hasQuote = this.capturedQuote.length > 0;
    const shortMain =
      this.capturedWord.length > 10 ? this.capturedWord.slice(0, 9) + "…" : this.capturedWord;
    const shortFs =
      this.capturedWord.length > 7 ? this.capturedWord.slice(0, 6) + "…" : this.capturedWord;

    if (this.quickTrBtn) this.quickTrBtn.toggleClass("is-ready", hasWord);
    if (this.fsQuickTrBtn) this.fsQuickTrBtn.toggleClass("is-ready", hasWord);

    if (this.addWordBtn) {
      this.addWordBtn.toggleClass("is-ready", hasWord);
      this.addWordBtn.setText(hasWord ? `+ «${shortMain}»` : "+ Слово");
    }
    if (this.fsAddWordBtn) {
      this.fsAddWordBtn.toggleClass("is-ready", hasWord);
      this.fsAddWordBtn.setText(hasWord ? `+ «${shortFs}»` : "+ Слово");
    }
    if (this.addQuoteBtn) this.addQuoteBtn.toggleClass("is-ready", hasQuote);
    if (this.fsAddQuoteBtn) this.fsAddQuoteBtn.toggleClass("is-ready", hasQuote);
  }

  openWordFlowWithExplicitWord(wordToSave: string, cfiToSave: string | null, sentenceEx: string) {
    if (!this.file) return;
    const folder = this.plugin.settings.booksFolder || "Знания/Книги";

      this.linkedBookNote = exactByProp[0];
      this.openAddWordModalWithRefresh(wordToSave, cfiToSave, sentenceEx, this.linkedBookNote, all);

    });

  turnPage(dir: "next" | "prev") {
    if (!this.rendition) return;
    this.recordReadingActivity();
    if (dir === "next") {
      recordPageTransitionForPace(this);
    } else {
      this.lastTransitionTimestamp = Date.now();
    }

    this.isResizing = false;
    if (this.resizeTimer) {
      window.clearTimeout(this.resizeTimer);
      this.resizeTimer = null;
    }

    this.clearCapturedSelection();

    const anim = this.plugin.settings.pageAnimation ?? "slide";

    if (anim !== "none" && this.boxEl) {
      if (this.animTimer) window.clearTimeout(this.animTimer);
      this.boxEl.removeClass("my-epub-anim-next", "my-epub-anim-prev", "my-epub-anim-fade");
      void this.boxEl.offsetWidth;

      if (anim === "fade") {
        this.boxEl.addClass("my-epub-anim-fade");
      } else {
        this.boxEl.addClass(dir === "next" ? "my-epub-anim-next" : "my-epub-anim-prev");
      }

      this.animTimer = window.setTimeout(() => {
        this.boxEl?.removeClass("my-epub-anim-next", "my-epub-anim-prev", "my-epub-anim-fade");
      }, 240);
    }

    if (dir === "next") {
      this.rendition.next();
    } else {
      this.rendition.prev();
    }
  }

  isCurrentPageBookmarked(): boolean {
    if (!this.file) return false;
    const list = this.plugin.settings.bookmarks?.[this.file.path] ?? [];
    if (this.currentPageNum) {
      return list.some((b) => b.page === this.currentPageNum);
    }
    return Boolean(this.stableCfi && list.some((b) => b.cfi === this.stableCfi));
  }

  triggerResize(force = false) {
    if (!this.boxEl) return;

    if (!force) {
      if (document.querySelector(".modal-container")) return;
      const activeTag = document.activeElement?.tagName;
      if (activeTag === "INPUT" || activeTag === "TEXTAREA" || activeTag === "SELECT") return;
    }

    const { clientWidth, clientHeight } = this.boxEl;
    if (clientWidth <= 0 || clientHeight <= 0) return;

    if (
      !force &&
      Math.abs(clientWidth - this.lastBoxWidth) < 3 &&
      Math.abs(clientHeight - this.lastBoxHeight) < 3
    ) {
      return;
    }

    const savedAnchor = this.stableCfi;
    this.isResizing = true;
    if (this.resizeTimer) window.clearTimeout(this.resizeTimer);

    this.resizeTimer = window.setTimeout(() => {
      if (!this.rendition || !this.boxEl) {
        this.isResizing = false;
        return;
      }
      const w = this.boxEl.clientWidth;
      const h = this.boxEl.clientHeight;
      if (w <= 0 || h <= 0) {
        this.isResizing = false;
        return;
      }

      this.lastBoxWidth = w;
      this.lastBoxHeight = h;

      this.rendition.resize(w, h);

      this.resizeTimer = window.setTimeout(async () => {
        if (!this.rendition) {
          this.isResizing = false;
          return;
        }
        if (savedAnchor) {
          this.stableCfi = savedAnchor;
          try {
            await this.rendition.display(savedAnchor);
          } catch {
            // Игнорируем
          }
        }
        window.setTimeout(() => {
          this.isResizing = false;
          if (this.file && this.stableCfi) {
            const rendExt = this.rendition as unknown as EpubRenditionExtended | null;
            const cur = rendExt?.currentLocation?.();
            if (cur?.start) captureChapterMetricsFromLoc(this, cur);
            this.plugin.settings.progress[this.file.path] = this.stableCfi;
            this.plugin.requestSaveSettings();
            this.updateProgressLabels(this.stableCfi);
          }
        }, 60);
      }, 120);
    }, 160);
  }

  setupResizeObserver() {
    this.resizeObserver?.disconnect();
    this.resizeObserver = new ResizeObserver(() => this.triggerResize(false));
    this.resizeObserver.observe(this.boxEl);
  }

  updateThemeBtnText() {

      mode === "sepia"
        ? "Режим: Сепия (нажми для режима Ночь)"
        : mode === "night"
          ? "Режим: Ночь (нажми для Авто)"
          : "Режим: Авто (нажми для Сепии)";
    this.themeBtn.setText(icon);
    this.themeBtn.setAttr("aria-label", label);
  }

    this.updateThemeBtnText();

    this.plugin.requestSaveSettings();
  }

      }, 60);

// B4: лёгкая нормализация окончаний (-s, -es, -ed, -ing, -ly) только для ключа поиска дублей
function lightStemToken(token: string): string {
  const w = token.trim();
  if (w.length < 4) return w;

  if (w.endsWith("ies") && w.length >= 5) {
    return w.slice(0, -3) + "y";
  }
  if (w.endsWith("ly") && w.length >= 5) {
    return w.slice(0, -2);
  }
  if (w.endsWith("ing") && w.length >= 6) {
    const stem = w.slice(0, -3);
    const last = stem.charAt(stem.length - 1);
    const prev = stem.charAt(stem.length - 2);
    if (stem.length >= 4 && last === prev && /[bdfgklmnprstvz]/.test(last)) {
      return stem.slice(0, -1);
    }
    return stem;
  }
  if (w.endsWith("ed") && w.length >= 5) {
    const stem = w.slice(0, -2);
    const last = stem.charAt(stem.length - 1);
    const prev = stem.charAt(stem.length - 2);
    if (stem.length >= 4 && last === prev && /[bdfgklmnprstvz]/.test(last)) {
      return stem.slice(0, -1);
    }
    return stem;
  }
  if (w.endsWith("es") && w.length >= 5) {
    if (/(ches|shes|xes|zes|sses)$/.test(w)) {
      return w.slice(0, -2);
    }
    return w.slice(0, -1);
  }
  if (w.endsWith("s") && !/(ss|us|is)$/.test(w)) {
    return w.slice(0, -1);
  }
  return w;
}

// B3: Защита сокращений (Mr., Dr., U.S., e.g. и инициалов) от ложного разрыва предложения
const ABBREV_END_REGEX =
  /(?:^|\s)(?:Mr|Mrs|Ms|Dr|Prof|Sr|Jr|St|Mt|vs|etc|fig|vol|ch|pp|no|e\.g|i\.e|U\.S|U\.K|Ph\.D|[A-Z])\.$/i;

function mergeAbbreviationSplits(rawSegments: string[]): string[] {
  const merged: string[] = [];
  for (const seg of rawSegments) {
    const trimmed = seg.trim();
    if (!trimmed) continue;
    const prev = merged[merged.length - 1];
    if (prev && ABBREV_END_REGEX.test(prev)) {
      merged[merged.length - 1] = `${prev} ${trimmed}`;
    } else {
      merged.push(trimmed);
    }
  }
  return merged;
}

function splitParagraphIntoSentences(paragraph: string): string[] {
  try {
    const IntlAny = typeof Intl !== "undefined" ? (Intl as any) : null;
    if (IntlAny && typeof IntlAny.Segmenter === "function") {
      const segmenter = new IntlAny.Segmenter("en", { granularity: "sentence" });
      const rawList = Array.from(segmenter.segment(paragraph), (item: any) =>
        String(item?.segment ?? "").trim()
      ).filter(Boolean);
      if (rawList.length > 0) {
        return mergeAbbreviationSplits(rawList);
      }
    }
  } catch {
    // Переходим к запасному варианту
  }

  // Запасной вариант с маскировкой точек внутри сокращений (Mr., Dr., U.S. и т.д.)
  const DOT_PLACEHOLDER = "\uE000";
  const masked = paragraph
    .replace(
      /\b(Mr|Mrs|Ms|Dr|Prof|Sr|Jr|St|Mt|vs|etc|fig|vol|ch|pp|no)\./gi,
      `$1${DOT_PLACEHOLDER}`
    )
    .replace(/\b(e\.g|i\.e|U\.S|U\.K|Ph\.D)\./gi, (m) => m.replace(/\./g, DOT_PLACEHOLDER))
    .replace(/\b([A-Z])\.(?=\s+[A-Z])/g, `$1${DOT_PLACEHOLDER}`);

  const matches = masked.match(/[^.!?]+[.!?]+["”»'’)]*|[^.!?]+$/g) || [masked];
  return matches
    .map((s) => s.replace(/\uE000/g, ".").trim())
    .filter(Boolean);
}

export function extractSentenceFromSelection(sel: Selection | null | undefined, word: string): string {
  if (!sel || sel.rangeCount === 0) return "";
  try {
    const range = sel.getRangeAt(0);
    let node: Node | null = range.startContainer;
    let blockEl: HTMLElement | null = null;

    while (node && node.nodeName !== "BODY" && node.nodeName !== "HTML") {
      if (node.nodeType === 1) {
        const tag = (node as HTMLElement).tagName.toUpperCase();
        if (/^(P|LI|BLOCKQUOTE|DD|DT|H1|H2|H3|H4|H5|H6|DIV|TD)$/.test(tag)) {
          blockEl = node as HTMLElement;
          break;
        }
      }
      node = node.parentNode;
    }

    const rawText = (blockEl ? blockEl.textContent : range.startContainer.textContent) || "";
    const paragraph = rawText.replace(/\s+/g, " ").trim();
    if (!paragraph) return "";

    const sentences = splitParagraphIntoSentences(paragraph);
    const cleanW = word.trim().toLowerCase();

    for (const s of sentences) {
      const trimmed = s.trim();
      if (cleanW && trimmed.toLowerCase().includes(cleanW)) {
        return cleanExample(trimmed);
      }
    }
    return cleanExample(sentences[0]?.trim() || "");
  } catch {
    return "";
  }
}

export function configureTranslationCache(app: App, pluginDir?: string) {

function resolveCacheAdapterAndPath(): { adapter: DataAdapter; path: string } | null {
  if (cacheAdapter && cacheFilePath) {
    return { adapter: cacheAdapter, path: cacheFilePath };
  }
  const globalApp: App | undefined =
    typeof window !== "undefined" ? (window as any).app : undefined;
  if (globalApp?.vault?.adapter) {
    configureTranslationCache(globalApp);
    if (cacheAdapter && cacheFilePath) {
      return { adapter: cacheAdapter, path: cacheFilePath };
    }
  }
  return null;
}

async function ensureTranslationCacheLoaded(): Promise<void> {
  if (cacheLoadedFromDisk) return;
  if (cacheLoadPromise) return cacheLoadPromise;

  cacheLoadPromise = (async () => {
    const resolved = resolveCacheAdapterAndPath();
    if (!resolved) {
      cacheLoadedFromDisk = true;
      return;
    }
    try {
      if (await resolved.adapter.exists(resolved.path)) {
        const raw = await resolved.adapter.read(resolved.path);
        const parsed = JSON.parse(raw);
        if (parsed && typeof parsed === "object") {
          const entries = Object.entries(parsed as Record<string, TranslationCacheRecord>);
          entries.sort((a, b) => (a[1]?.updatedAt ?? 0) - (b[1]?.updatedAt ?? 0));
          for (const [k, v] of entries) {
            if (v && Array.isArray(v.groups) && v.groups.length > 0) {
              translationCache.set(k, {
                groups: v.groups,
                updatedAt: typeof v.updatedAt === "number" ? v.updatedAt : Date.now(),
              });
            }
          }
          pruneTranslationCacheInMemory();
        }
      }
    } catch {
      // Игнорируем ошибку чтения кэша переводов
    } finally {
      cacheLoadedFromDisk = true;
      cacheLoadPromise = null;
    }
  })();

  return cacheLoadPromise;
}

function pruneTranslationCacheInMemory() {
  if (translationCache.size <= MAX_TRANSLATION_CACHE) return;
  const entries = Array.from(translationCache.entries());
  entries.sort((a, b) => a[1].updatedAt - b[1].updatedAt);
  const toRemoveCount = translationCache.size - MAX_TRANSLATION_CACHE;
  for (let i = 0; i < toRemoveCount; i++) {
    const item = entries[i];
    if (item) translationCache.delete(item[0]);
  }
}

function scheduleSaveTranslationCache() {
  if (cacheSaveTimer !== null) window.clearTimeout(cacheSaveTimer);
  cacheSaveTimer = window.setTimeout(async () => {
    cacheSaveTimer = null;
    const resolved = resolveCacheAdapterAndPath();
    if (!resolved) return;
    try {
      pruneTranslationCacheInMemory();
      const obj: Record<string, TranslationCacheRecord> = {};
      for (const [k, v] of translationCache.entries()) {
        obj[k] = v;
      }
      await resolved.adapter.write(resolved.path, JSON.stringify(obj));
    } catch {
      // Игнорируем ошибку фоновой записи кэша
    }
  }, TRANSLATION_CACHE_DEBOUNCE_MS);
}

function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
  return Promise.race([
    p,
    new Promise<T>((_, rej) =>
      setTimeout(() => rej(new Error(`нет ответа за ${ms / 1000} с`)), ms)
    ),
  ]);
}

function parseGoogleData(d: any): TranslationGroup[] {
  const groups: TranslationGroup[] = [];
  if (Array.isArray(d?.[1])) {
    d[1].forEach((e: any) => {
      const terms = (Array.isArray(e?.[1]) ? e[1] : []).filter(
        (t: any) => typeof t === "string" && t.trim()
      );
      if (terms.length) {
        groups.push({
          pos: String(e[0] || "перевод"),
          terms: terms.slice(0, TRANSLATE_PER_POS).map((t: string) => t.trim()),
        });
      }
    });
  }
  if (!groups.length && Array.isArray(d?.dict)) {
    d.dict.forEach((entry: any) => {
      const terms = (Array.isArray(entry?.terms) ? entry.terms : []).filter(
        (t: any) => typeof t === "string" && t.trim()
      );
      if (terms.length) {
        groups.push({
          pos: String(entry.pos || "перевод"),
          terms: terms.slice(0, TRANSLATE_PER_POS).map((t: string) => t.trim()),
        });
      }
    });
  }
  if (!groups.length && Array.isArray(d?.sentences)) {
    const main = d.sentences.map((s: any) => s?.trans).filter(Boolean).join("").trim();
    if (main) groups.push({ pos: "перевод", terms: [main] });
  }
  if (!groups.length && Array.isArray(d?.[0])) {
    const main = d[0].map((s: any) => s?.[0]).filter(Boolean).join("").trim();
    if (main) groups.push({ pos: "перевод", terms: [main] });
  }
  return groups;
}

export async function fetchTranslations(word: string): Promise<TranslationGroup[]> {
  const cleanWord = String(word ?? "").trim();
  if (!cleanWord) throw new Error("пустой запрос");

  await ensureTranslationCacheLoaded();

  const cacheKey = cleanWord.toLowerCase();
  const cached = translationCache.get(cacheKey);
  if (cached && cached.groups.length > 0) {
    cached.updatedAt = Date.now();
    scheduleSaveTranslationCache();
    return cached.groups;
  }

  const q = encodeURIComponent(cleanWord);
  const endpoints = [
    `https://clients5.google.com/translate_a/t?client=dict-chrome-ex&sl=en&tl=ru&q=${q}`,
    `https://translate.googleapis.com/translate_a/single?client=gtx&dt=t&dt=bd&dj=1&sl=en&tl=ru&q=${q}`,
    `https://translate.google.com/translate_a/single?client=at&sl=en&tl=ru&dt=t&dt=bd&q=${q}`,
    `https://translate.googleapis.com/translate_a/single?client=gtx&sl=en&tl=ru&dt=t&dt=bd&q=${q}`,
  ];

  let lastErr: any = null;
  for (const url of endpoints) {
    try {
      const r = await withTimeout(requestUrl({ url, headers: DESKTOP_HEADERS }), TRANSLATE_TIMEOUT_MS);
      const data = typeof r.json === "object" && r.json !== null ? r.json : JSON.parse(r.text);
      const groups = parseGoogleData(data);
      if (groups.length) {
        translationCache.set(cacheKey, { groups, updatedAt: Date.now() });
        pruneTranslationCacheInMemory();
        scheduleSaveTranslationCache();
        return groups;
      }
    } catch (e) {
      lastErr = e;
    }
  }

  try {
    const fallbackUrl = `https://api.mymemory.translated.net/get?q=${q}&langpair=en%7Cru`;
    const r = await withTimeout(
      requestUrl({ url: fallbackUrl, headers: DESKTOP_HEADERS }),
      TRANSLATE_TIMEOUT_MS
    );
    const data = typeof r.json === "object" && r.json !== null ? r.json : JSON.parse(r.text);
    const termsSet = new Set<string>();

    const mainText = data?.responseData?.translatedText;
    if (typeof mainText === "string" && mainText.trim() && mainText.toLowerCase() !== cacheKey) {
      termsSet.add(mainText.trim().toLowerCase());
    }
    if (Array.isArray(data?.matches)) {
      data.matches.forEach((m: any) => {
        const t = String(m?.translation ?? "").trim();
        if (t && t.toLowerCase() !== cacheKey && !/[a-zA-Z]{3,}/.test(t)) {
          termsSet.add(t.toLowerCase());
        }
      });
    }

    const terms = [...termsSet].slice(0, TRANSLATE_PER_POS);
    if (terms.length) {
      const groups = [{ pos: "перевод", terms }];
      translationCache.set(cacheKey, { groups, updatedAt: Date.now() });
      pruneTranslationCacheInMemory();
      scheduleSaveTranslationCache();
      return groups;
    }
  } catch (e) {
    lastErr = e;
  }

  throw lastErr || new Error("сервисы перевода временно недоступны");
}

interface ExistingWordCard {
  lineNumber: number;
  front: string;
  word: string;
  back: string;
  key: string;
}

function parseExistingWordsForDuplicates(content: string): ExistingWordCard[] {
  const lines = String(content ?? "").split(/\r?\n/);
  const result: ExistingWordCard[] = [];

  for (let i = 0; i < lines.length; i++) {
    const line = lines[i] ?? "";
    const srIdx = line.indexOf("<!--SR:");
    const cardPart = srIdx === -1 ? line : line.slice(0, srIdx);
    if (!/:{2,3}/.test(cardPart) || cardPart.trimStart().startsWith("#")) continue;

    const parts = cardPart.split(/:{2,3}/);
    const rawFront = (parts[0] ?? "").replace(/^[-*+]\s+/, "").trim();
    const rawBackJoined = parts.slice(1).join(" ").trim();

    const exIdx = rawFront.indexOf(EXAMPLE_SEP);
    const word = exIdx === -1 ? rawFront : rawFront.slice(0, exIdx).trim();

    let cleanBack = rawBackJoined;
    const inlineMatch = rawBackJoined.match(
      /^(.*?)\s*\((?:источник|книга)::\s*(\[\[[^\]]+\]\])\)\s*(?![\s\S])/i
    );
    if (inlineMatch?.[1] !== undefined) {
      cleanBack = inlineMatch[1].trim();
    } else {
      const dotMatch = rawBackJoined.match(/^(.*?)\s*(?:·|\|)\s*(\[\[[^\]]+\]\])\s*(?![\s\S])/);
      if (dotMatch?.[1] !== undefined) {
        cleanBack = dotMatch[1].trim();
      }
    }

    result.push({
      lineNumber: i + 1,
      front: rawFront,
      word,
      back: cleanBack,
      key: wordKey(word),
    });
  }

  return result;
}

// B2: Парсер существующих цитат для проверки дублей внутри выбранной книги
interface ExistingQuoteRecord {
  lineNumber: number;
  text: string;
  normKey: string;
  bookTarget: string | null;
  bookTitleNorm: string;
}

function parseExistingQuotesForDuplicates(content: string): ExistingQuoteRecord[] {
  const lines = String(content ?? "").split(/\r?\n/);
  const result: ExistingQuoteRecord[] = [];

  for (let i = 0; i < lines.length; i++) {
    const line = (lines[i] ?? "").trim();
    if (!line.startsWith("-") || !line.includes("цитата::")) continue;

    const fields: Record<string, string> = {};
    const fieldRegex = /\(([а-яА-Яa-zA-Z0-9_]+)::\s*([^)]+)\)/g;
    for (const m of line.matchAll(fieldRegex)) {
      const k = m[1]?.trim();
      const v = m[2]?.trim();
      if (k && v) fields[k] = v;
    }

    const rawText = fields["цитата"] ? decodeInline(fields["цитата"]) : "";
    if (!rawText) continue;

    const linkTarget =
      extractLinkTarget(fields["книга_ссылка"]) || extractLinkTarget(fields["книга"]);
    const bookTitleNorm = normStr(decodeInline(fields["книга"] || ""));

    result.push({
      lineNumber: i + 1,
      text: rawText,
      normKey: normText(rawText),
      bookTarget: linkTarget,
      bookTitleNorm,
    });
  }

  return result;
}

function isQuoteInSelectedBook(q: ExistingQuoteRecord, book: BookNoteInfo | null): boolean {
  if (!book) return false;
  if (q.bookTarget) {
    const cleanTarget = normStr(q.bookTarget.replace(/\.md$/i, ""));
    if (
      cleanTarget === normStr(book.fileName) ||
      cleanTarget === normStr(book.title) ||
      normStr(q.bookTarget) === normStr(book.path)
    ) {
      return true;
    }
  }
  return q.bookTitleNorm !== "" && q.bookTitleNorm === normStr(book.title);
}

function safeCloseModal(modal: Modal) {
  try {
    (modal as any).shouldRestoreSelection = false;
    modal.close();
  } catch (e) {
    console.log("Safe modal close:", e);
  }
}

export class TocModal extends Modal {
  toc: NavItem[];
  onSelect: (href: string) => void;

  constructor(app: App, toc: NavItem[], onSelect: (href: string) => void) {
    super(app);
    (this as any).shouldRestoreSelection = false;
    this.toc = toc;
    this.onSelect = onSelect;
  }

  onOpen() {
    (this as any).shouldRestoreSelection = false;
    const { contentEl } = this;
    contentEl.empty();
    contentEl.addClass("my-epub-toc-modal");
    contentEl.createEl("h3", { text: "📑 Оглавление" });

    if (!this.toc || this.toc.length === 0) {
      contentEl.createDiv({ cls: "my-epub-toc-empty", text: "В этой книге нет оглавления." });
      return;
    }

    const list = contentEl.createDiv("my-epub-toc-list");
    this.renderItems(list, this.toc, 0);
  }

  renderItems(container: HTMLElement, items: NavItem[], level: number) {
    for (const item of items) {
      const row = container.createDiv({ cls: "my-epub-toc-item" });
      row.style.paddingLeft = `${0.75 + level * 1.25}rem`;
      row.setText((item.label || "Без названия").trim());
      row.onclick = () => {
        this.onSelect(item.href);
        safeCloseModal(this);
      };
      if (item.subitems && item.subitems.length > 0) {
        this.renderItems(container, item.subitems, level + 1);
      }
    }
  }

  onClose() {
    this.contentEl.empty();
  }
}

export class JumpToPageModal extends Modal {
  currentPage: number;
  totalPages: number;
  chapterInfoText: string;
  onJumpPage: (page: number) => void;

  constructor(
    app: App,
    currentPage: number,
    totalPages: number,
    chapterInfoText: string,
    onJumpPage: (page: number) => void
  ) {
    super(app);
    (this as any).shouldRestoreSelection = false;
    this.currentPage = currentPage;
    this.totalPages = totalPages;
    this.chapterInfoText = chapterInfoText;
    this.onJumpPage = onJumpPage;
  }

  onOpen() {
    (this as any).shouldRestoreSelection = false;
    const { contentEl } = this;
    contentEl.empty();
    contentEl.addClass("my-epub-modal");

    contentEl.createEl("h3", { text: "🧭 Переход по книге" });

    if (this.chapterInfoText) {
      contentEl.createDiv({
        cls: "my-epub-modal-sub",
        text: this.chapterInfoText,
      });
    }

    const row = contentEl.createDiv("my-epub-modal-row");
    const pageInput = row.createEl("input", {
      cls: "my-epub-input-en",
      type: "number",
      attr: {
        min: "1",
        max: String(this.totalPages),
        step: "1",
        placeholder: `Страница (1 – ${this.totalPages})`,
      },
      value: String(this.currentPage),
    });

    const goBtn = row.createEl("button", {
      cls: "mod-cta my-epub-modal-btn",
      text: "Перейти",
    });

    const sliderWrap = contentEl.createDiv("my-epub-modal-section");
    const pctNow = Math.round((this.currentPage / Math.max(1, this.totalPages)) * 100);
    const sliderLabel = sliderWrap.createDiv({
      cls: "my-epub-modal-label",
      text: `Позиция: стр. ${this.currentPage} из ${this.totalPages} (${pctNow}%)`,
    });

    const rangeInput = sliderWrap.createEl("input", {
      type: "range",
      attr: {
        min: "1",
        max: String(this.totalPages),
        step: "1",
        style: "width: 100%; cursor: pointer;",
      },
      value: String(this.currentPage),
    });

    rangeInput.oninput = () => {
      const p = Math.max(1, Math.min(this.totalPages, Number(rangeInput.value) || 1));
      const pct = Math.round((p / Math.max(1, this.totalPages)) * 100);
      pageInput.value = String(p);
      sliderLabel.setText(`Позиция: стр. ${p} из ${this.totalPages} (${pct}%)`);
    };

    pageInput.oninput = () => {
      const p = Math.max(1, Math.min(this.totalPages, Number(pageInput.value) || 1));
      const pct = Math.round((p / Math.max(1, this.totalPages)) * 100);
      rangeInput.value = String(p);
      sliderLabel.setText(`Позиция: стр. ${p} из ${this.totalPages} (${pct}%)`);
    };

    const doJump = () => {
      const target = Math.max(1, Math.min(this.totalPages, Number(pageInput.value) || 1));
      this.onJumpPage(target);
      safeCloseModal(this);
    };

    goBtn.onclick = doJump;
    pageInput.onkeydown = (e) => {
      if (e.key === "Enter") {
        e.preventDefault();
        doJump();
      }
    };

    window.setTimeout(() => {
      pageInput.focus();
      pageInput.select();
    }, 40);
  }

  onClose() {
    this.contentEl.empty();
  }
}

// --- 2. Привязка или создание заметки книги ---
export class BookLinkModal extends Modal {
  epubFile: TFile;
  booksFolder: string;
  candidates: BookNoteInfo[];
  allBooks: BookNoteInfo[];
  defaultTitle: string;
  defaultAuthor: string;
  currentPage: number | null;
  totalPages: number | null;
  currentPct: number | null;
  onResolved: (bookNote: BookNoteInfo) => void;

  constructor(
    app: App,
    epubFile: TFile,
    booksFolder: string,
    candidates: BookNoteInfo[],
    allBooks: BookNoteInfo[],
    defaultTitle: string,
    defaultAuthor: string,
    currentPage: number | null,
    totalPages: number | null,
    currentPct: number | null,
    onResolved: (bookNote: BookNoteInfo) => void
  ) {
    super(app);
    (this as any).shouldRestoreSelection = false;
    this.epubFile = epubFile;
    this.booksFolder = booksFolder;
    this.candidates = candidates;
    this.allBooks = allBooks;
    this.defaultTitle = defaultTitle;
    this.defaultAuthor = defaultAuthor;
    this.currentPage = currentPage;
    this.totalPages = totalPages;
    this.currentPct = currentPct;
    this.onResolved = onResolved;
  }

  onOpen() {
    (this as any).shouldRestoreSelection = false;
    const { contentEl } = this;
    contentEl.empty();
    contentEl.addClass("my-epub-modal");

    contentEl.createEl("h3", { text: "📖 Привязка заметки книги" });
    contentEl.createDiv({
      cls: "my-epub-modal-sub",
      text: `Файл: ${this.epubFile.path}`,
    });

    const existingBox = contentEl.createDiv("my-epub-modal-section");
    existingBox.createDiv({
      cls: "my-epub-modal-label",
      text:
        this.candidates.length > 0
          ? "Найдена похожая заметка в «" + this.booksFolder + "»:"
          : "Выбрать существующую заметку из «" + this.booksFolder + "»:",
    });

    if (this.allBooks.length > 0) {
      const row = existingBox.createDiv("my-epub-modal-row");
      const select = row.createEl("select", { cls: "dropdown my-epub-input-flex" });

      const sorted = [...this.allBooks].sort((a, b) => {
        const aCand = this.candidates.some((c) => c.path === a.path) ? 0 : 1;
        const bCand = this.candidates.some((c) => c.path === b.path) ? 0 : 1;
        if (aCand !== bCand) return aCand - bCand;
        const aRead = a.status === "Читаю" ? 0 : 1;
        const bRead = b.status === "Читаю" ? 0 : 1;
        return aRead - bRead || a.title.localeCompare(b.title, "ru", { sensitivity: "base" });
      });

      sorted.forEach((b, idx) => {
        const isCand = this.candidates.some((c) => c.path === b.path);
        const prefix = isCand ? "★ " : b.status === "Читаю" ? "📖 " : "";
        const opt = select.createEl("option", {
          text: `${prefix}${b.title}${b.author ? " — " + b.author : ""} [${b.status}]`,
          value: b.path,
        });
        if (idx === 0) opt.selected = true;
      });

      const linkBtn = row.createEl("button", {
        cls: "mod-cta my-epub-modal-btn",
        text: "Привязать выбранную",
      });

      const confirmConflictBox = existingBox.createDiv({
        cls: "my-epub-conflict-box",
      });
      confirmConflictBox.style.display = "none";

      linkBtn.onclick = async () => {
        const chosen = this.allBooks.find((b) => b.path === select.value);
        if (!chosen) return;

        const existingVal = chosen.bookPropRaw;
        const targetWiki = `[[${this.epubFile.path}]]`;

        if (
          existingVal &&
          existingVal !== targetWiki &&
          extractLinkTarget(existingVal) !== this.epubFile.path
        ) {
          confirmConflictBox.empty();
          confirmConflictBox.style.display = "flex";
          confirmConflictBox.createDiv({
            text: `⚠️ В заметке «${chosen.title}» свойство «книга» уже занято: ${existingVal}. Перезаписать его ссылкой на этот EPUB?`,
          });
          const btnRow = confirmConflictBox.createDiv("my-epub-modal-row");
          const yesBtn = btnRow.createEl("button", {
            cls: "mod-cta my-epub-modal-btn",
            text: "Да, перезаписать",
          });
          const noBtn = btnRow.createEl("button", {
            cls: "my-epub-modal-btn",
            text: "Только для этого сеанса (не менять YAML)",
          });

          yesBtn.onclick = async () => {
            await this.writeBookLinkToFrontmatter(chosen);
          };
          noBtn.onclick = () => {
            this.onResolved(chosen);
            safeCloseModal(this);
          };
          return;
        }

        await this.writeBookLinkToFrontmatter(chosen);
      };
    } else {
      existingBox.createDiv({
        cls: "my-epub-modal-sub",
        text: "В папке ещё нет заметок книг.",
      });
    }

    const createBox = contentEl.createDiv("my-epub-modal-section");
    createBox.createDiv({
      cls: "my-epub-modal-label",
      text: "Или создать новую заметку книги в «" + this.booksFolder + "»:",
    });

    const grid = createBox.createDiv("my-epub-modal-grid");
    const titleInp = grid.createEl("input", {
      type: "text",
      placeholder: "Название книги *",
      value: this.defaultTitle,
    });
    const authorInp = grid.createEl("input", {
      type: "text",
      placeholder: "Автор",
      value: this.defaultAuthor,
    });
    const genreInp = grid.createEl("input", {
      type: "text",
      placeholder: "Жанр (необязательно)",
    });

    const createBtnRow = createBox.createDiv("my-epub-modal-row-end");
    const createBtn = createBtnRow.createEl("button", {
      cls: "my-epub-modal-btn",
      text: "+ Создать заметку книги",
    });

    createBtn.onclick = async () => {
      const title = titleInp.value.trim();
      const author = authorInp.value.trim();
      const genre = genreInp.value.trim();

      if (!title) {
        new Notice("Укажи название книги.");
        return;
      }

      createBtn.disabled = true;
      let resolvedInfo: BookNoteInfo | null = null;
      try {
        await ensureFolderExists(this.app, this.booksFolder);
        const safeBase = title.replace(/[\\/:*?"<>|#^\[\]]/g, "").trim();
        if (!safeBase) throw new Error("Некорректное название книги.");

        const folderPrefix = this.booksFolder.replace(/\/+$/, "");
        let targetPath = `${folderPrefix}/${safeBase}.md`;
        if (this.app.vault.getAbstractFileByPath(targetPath)) {
          const safeAuthor = author.replace(/[\\/:*?"<>|#^\[\]]/g, "").trim();
          if (safeAuthor) {
            targetPath = `${folderPrefix}/${safeBase} (${safeAuthor}).md`;
          }
          if (this.app.vault.getAbstractFileByPath(targetPath)) {
            throw new Error(`Заметка «${targetPath}» уже существует. Выбери её в списке выше.`);
          }
        }

        const wikiLink = `[[${this.epubFile.path}]]`;
        const todayISO = getTodayISO();
        const hasTotal = this.totalPages !== null && this.totalPages > 0;
        const hasRead = this.currentPage !== null && this.currentPage >= 0;
        const hasPct = this.currentPct !== null && this.currentPct >= 0;

        const yamlLines = [
          "---",
          `название: ${JSON.stringify(title)}`,
          `автор: ${JSON.stringify(author)}`,
          `жанр: ${JSON.stringify(genre)}`,
          `статус: "Читаю"`,
          `страниц: ${hasTotal ? this.totalPages : ""}`,
          `прочитано: ${hasRead ? this.currentPage : ""}`,
          `прогресс: ${hasPct ? this.currentPct : ""}`,
          `последнее_чтение: ${JSON.stringify(todayISO)}`,
          `книга: ${JSON.stringify(wikiLink)}`,
          "---",
          "",
          `# ${title}`,
          "",
        ];

        const created = await this.app.vault.create(targetPath, yamlLines.join("\n"));
        resolvedInfo = {
          file: created,
          path: created.path,
          fileName: created.basename,
          title,
          author,
          status: "Читаю",
          bookPropRaw: wikiLink,
          linkedEpubPath: this.epubFile.path,
        };
        new Notice(`Создана и привязана книга «${title}»`);
      } catch (e: any) {
        new Notice("Ошибка: " + (e?.message || e));
        createBtn.disabled = false;
        return;
      }

      if (resolvedInfo) {
        this.onResolved(resolvedInfo);
        safeCloseModal(this);
      }
    };
  }

  async writeBookLinkToFrontmatter(chosen: BookNoteInfo) {
    const wikiLink = `[[${this.epubFile.path}]]`;
    const todayISO = getTodayISO();
    try {
      await this.app.fileManager.processFrontMatter(chosen.file, (fm) => {
        fm["книга"] = wikiLink;
        if (this.totalPages !== null && this.totalPages > 0) {
          fm["страниц"] = this.totalPages;
        }
        if (this.currentPage !== null && this.currentPage > 0) {
          fm["прочитано"] = this.currentPage;
        }
        if (this.currentPct !== null && this.currentPct >= 0) {
          fm["прогресс"] = this.currentPct;
        }
        fm["последнее_чтение"] = todayISO;
        if (!fm["статус"] || fm["статус"] === "В планах") {
          fm["статус"] = "Читаю";
        }
      });
      chosen.bookPropRaw = wikiLink;
      chosen.linkedEpubPath = this.epubFile.path;
      new Notice(`Книга привязана к заметке «${chosen.title}»`);
    } catch (e: any) {
      new Notice("Ошибка записи YAML: " + (e?.message || e));
      return;
    }
    this.onResolved(chosen);
    safeCloseModal(this);
  }

  onClose() {
    this.contentEl.empty();
  }

        const freshContent = await this.app.vault.read(targetWordsFile);

        if (same.length) {
          showDup(same);
          const exact = same.some((c) => normText(c.back) === normText(ru));
          if (exact) {
            new Notice("Такая карточка с этим переводом уже есть — дубликат не добавлен.");
            addBtn.disabled = false;
            return;
          }
          const pk = `${k}|${normText(ru)}`;

            pendingKey = pk;
            addBtn.setText("Всё равно добавить");
            new Notice(
              "Слово уже есть с другим переводом. Нажми ещё раз, если хочешь добавить ещё одну карточку."
            );
            addBtn.disabled = false;
            return;
          }
        }

        const front = ex ? `${en}${EXAMPLE_SEP}${ex}` : en;
        let backWithSource = ru;
        if (chosenBook) {
          const cleanCfi =
            this.capturedCfi && !/[\[\]|#]/.test(this.capturedCfi) ? this.capturedCfi.trim() : "";
          const wiki = cleanCfi

            : `[[${chosenBook.fileName}]]`;
          backWithSource = `${ru} · ${wiki}`;
        }

        const prefix = freshContent.length > 0 && !freshContent.endsWith("\n") ? "\n" : "";
        await this.app.vault.append(targetWordsFile, `${prefix}${front} ::: ${backWithSource}\n`);

        savedSuccessfully = true;
        noticeText = `✓ Добавлено: ${front} ::: ${ru}${chosenBook ? " (" + chosenBook.title + ")" : ""}`;
      } catch (e: any) {
        new Notice("Ошибка добавления: " + (e?.message || e));
        addBtn.disabled = false;
        return;
      }

        new Notice(noticeText);
        safeCloseModal(this);
      }
    };

    addBtn.onclick = submitWord;

        void submitWord();
      }
    };

      if (e.key === "Enter") {
        e.preventDefault();
        void submitWord();
      }
    };

    if (this.initialWord) {
      void doTranslate();
    } else {
      window.setTimeout(() => enInput.focus(), 40);
    }
  }

    if (dest instanceof TFile) return dest;
    return null;

    this.contentEl.empty();

  if (typeof s.hyphens !== "boolean") s.hyphens = false;
  if (s.pageAnimation !== "slide" && s.pageAnimation !== "fade" && s.pageAnimation !== "none") {
    s.pageAnimation = "slide";
  }
  if (s.barPosition !== "bottom" && s.barPosition !== "top") s.barPosition = "bottom";
  return s;
}

export function hashBookPathToFileName(bookPath: string): string {
  let h = 2166136261;
  for (let i = 0; i < bookPath.length; i++) {
    h ^= bookPath.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  const hex = (h >>> 0).toString(16).padStart(8, "0");
  return `loc-${hex}.json`;
}

export function bindActionButton(
  view: EpubView,
  btn: HTMLElement,
  handler: () => void
) {
  let lastTrigger = 0;
  const trigger = (e: Event) => {
    e.preventDefault();
    e.stopPropagation();
    const now = Date.now();
    if (now - lastTrigger < 350) return;
    lastTrigger = now;
    view.recordReadingActivity();
    syncSelectionFromAllIframes(view);
    handler();
  };
  btn.addEventListener("touchstart", trigger, { passive: false });
  btn.addEventListener("mousedown", (e) => e.preventDefault());
  btn.addEventListener("click", trigger);
}

export function syncSelectionFromAllIframes(view: EpubView, cfiOverride?: string) {
  try {
    const rendExt = view.rendition as unknown as EpubRenditionExtended | null;
    const contentsList = rendExt?.getContents?.() || [];
    for (const c of contentsList) {
      const win = c.window;
      const doc = c.document;
      const sel = win?.getSelection?.() || doc?.getSelection?.();
      const raw = sel?.toString() || "";
      if (raw.trim().length > 0) {
        saveCapturedText(view, raw, cfiOverride, c, sel);
        return;
      }
    }

    const iframes = view.boxEl ? Array.from(view.boxEl.querySelectorAll("iframe")) : [];
    for (const frame of iframes) {
      const doc = frame.contentDocument;
      const win = frame.contentWindow;
      const sel = win?.getSelection?.() || doc?.getSelection?.();
      const raw = sel?.toString() || "";
      if (raw.trim().length > 0) {
        saveCapturedText(view, raw, cfiOverride, undefined, sel);
        return;
      }
    }
  } catch {
    // Игнорируем
  }
}

export function saveCapturedText(
  view: EpubView,
  raw: string,
  cfiOverride?: string,
  contentsObj?: EpubContentsExtended,
  selObj?: Selection | null
) {
  const cleanedWord = cleanSelectedWord(raw);
  const cleanedQuote = cleanSelectedQuote(raw);
  if (!cleanedWord && !cleanedQuote) return;

  view.recordReadingActivity();
  view.capturedWord = cleanedWord;
  view.capturedQuote = cleanedQuote;
  view.lastSelectionAt = Date.now();

  const sentence = extractSentenceFromSelection(selObj, cleanedWord);
  if (sentence) {
    view.extractedSentence = sentence;
  }

  if (cfiOverride) {
    view.capturedCfi = cfiOverride;
  } else if (
    contentsObj &&
    selObj &&
    selObj.rangeCount > 0 &&
    typeof contentsObj.cfiFromRange === "function"
  ) {
    try {
      const range = selObj.getRangeAt(0);
      const computedCfi = contentsObj.cfiFromRange(range);
      if (computedCfi) view.capturedCfi = computedCfi;
    } catch {
      view.capturedCfi = view.stableCfi;
    }
  } else if (!view.capturedCfi) {
    view.capturedCfi = view.stableCfi;
  }

  view.updateSelectionButtonsState();
}

export function attachInternalLinkInterceptor(view: EpubView, doc: Document) {
  doc.addEventListener(
    "click",
    (ev: MouseEvent) => {
      const target = ev.target as HTMLElement | null;
      const anchor = target?.closest?.("a[href]") as HTMLAnchorElement | null;
      if (!anchor) return;

      const rawHref = anchor.getAttribute("href")?.trim() || "";
      if (!rawHref) return;

      if (/^(https?:|mailto:|tel:)/i.test(rawHref)) {
        ev.preventDefault();
        ev.stopPropagation();
        window.open(rawHref, "_blank", "noopener,noreferrer");
        return;
      }

      view.pushNavHistory();
    },
    { capture: true }
  );
}

export function getRelativeTapRatio(
  view: EpubView,
  target: Document | HTMLElement,
  clientX: number
): number {
  const wrapRect = view.wrapEl?.getBoundingClientRect();
  const wrapWidth = wrapRect?.width || window.innerWidth || 1;
  const wrapLeft = wrapRect?.left || 0;

  if (target instanceof Document) {
    const frameEl = target.defaultView?.frameElement as HTMLElement | null;
    if (frameEl) {
      const frameRect = frameEl.getBoundingClientRect();
      const screenClientX = frameRect.left + clientX;
      return (screenClientX - wrapLeft) / wrapWidth;
    }
  }

  return (clientX - wrapLeft) / wrapWidth;
}

export function attachTouchHandlers(
  view: EpubView,
  target: Document | HTMLElement,
  isInsideIframe: boolean,
  contentsObj?: EpubContentsExtended
) {
  let startX = 0;
  let startY = 0;
  let lastX = 0;
  let lastY = 0;
  let startClientX = 0;
  let startTime = 0;
  let isSwiping = false;
  let active = false;
  let hadSelectionOnStart = false;

  const getDocSelectionString = (): string => {
    if (!isInsideIframe) return "";
    const doc = target as Document;
    const win = doc.defaultView;
    return (
      win?.getSelection?.()?.toString() ||
      doc.getSelection?.()?.toString() ||
      ""
    ).trim();
  };

  const onTouchStart = (ev: Event) => {
    const e = ev as TouchEvent;
    const touch = e.changedTouches?.[0] ?? e.touches?.[0];
    if (!touch) return;
    view.recordReadingActivity();
    startX = touch.screenX;
    startY = touch.screenY;
    lastX = touch.screenX;
    lastY = touch.screenY;
    startClientX = touch.clientX;
    startTime = Date.now();
    isSwiping = false;
    active = true;
    hadSelectionOnStart = getDocSelectionString().length > 0;
  };

  const onTouchMove = (ev: Event) => {
    if (!active) return;

    const currentSel = getDocSelectionString();
    if (hadSelectionOnStart || currentSel.length > 0) {
      if (currentSel.length > 0) {
        const doc = target as Document;
        const sel = doc.defaultView?.getSelection?.() || doc.getSelection?.();
        saveCapturedText(view, currentSel, undefined, contentsObj, sel);
      }
      return;
    }

    const e = ev as TouchEvent;
    const touch = e.touches?.[0] ?? e.changedTouches?.[0];
    if (!touch) return;
    lastX = touch.screenX;
    lastY = touch.screenY;
    const dx = lastX - startX;
    const dy = lastY - startY;

    if (Math.abs(dx) > 12 && Math.abs(dx) > Math.abs(dy) * 1.2) {
      isSwiping = true;
      if (e.cancelable) e.preventDefault();
      e.stopPropagation();
    }
  };

  const finishTouch = (ev: Event) => {
    if (!active) return;
    active = false;

    const selAfter = getDocSelectionString();
    if (selAfter.length > 0) {
      const doc = target as Document;
      const sel = doc.defaultView?.getSelection?.() || doc.getSelection?.();
      saveCapturedText(view, selAfter, undefined, contentsObj, sel);
      return;
    }

    if (hadSelectionOnStart || Date.now() - view.lastSelectionAt < 450) {
      return;
    }

    const e = ev as TouchEvent;
    const touch = e.changedTouches?.[0];
    const endX = touch ? touch.screenX : lastX;
    const endY = touch ? touch.screenY : lastY;
    const endClientX = touch ? touch.clientX : startClientX;
    const dx = endX - startX;
    const dy = endY - startY;
    const dt = Date.now() - startTime;

    const now = Date.now();
    if (now - view.lastGestureTime < 80) return;

    if (dt < 800 && Math.abs(dx) > 35 && Math.abs(dx) > Math.abs(dy) * 1.2) {
      view.lastGestureTime = now;
      if (dx < 0) view.turnPage("next");
      else view.turnPage("prev");
      return;
    }

    if (
      e.type === "touchend" &&
      !isSwiping &&
      dt < 350 &&
      Math.abs(dx) < 15 &&
      Math.abs(dy) < 15
    ) {
      const el = e.target as HTMLElement | null;
      if (el?.closest && el.closest("a, button, .my-epub-fs-bar, .my-epub-back-pill")) {
        return;
      }

      view.lastGestureTime = now;
      const ratio = getRelativeTapRatio(view, target, endClientX);
      if (ratio < 0.26) {
        view.turnPage("prev");
      } else if (ratio > 0.74) {
        view.turnPage("next");
      } else {
        view.toggleFullscreen();
      }
    }
  };

  target.addEventListener("touchstart", onTouchStart, { passive: true });
  target.addEventListener("touchmove", onTouchMove, { passive: false });
  target.addEventListener("touchend", finishTouch, { passive: true });
  target.addEventListener("touchcancel", finishTouch, { passive: true });

  if (!isInsideIframe) {
    view.cleanupListeners.push(() => {
      target.removeEventListener("touchstart", onTouchStart);
      target.removeEventListener("touchmove", onTouchMove);
      target.removeEventListener("touchend", finishTouch);
      target.removeEventListener("touchcancel", finishTouch);
    });
  }
}

export function setupViewGesturesAndSelection(view: EpubView) {
  if (!view.rendition) return;

  view.rendition.on("selected", (cfiRange: string, contents: EpubContentsExtended) => {
    try {
      const sel =
        contents?.window?.getSelection?.() || contents?.document?.getSelection?.();
      const raw = sel?.toString() || "";
      if (raw.trim()) {
        saveCapturedText(view, raw, cfiRange, contents, sel);
      }
    } catch {
      // Игнорируем
    }
  });

  view.rendition.hooks.content.register((contents: EpubContentsExtended) => {
    const doc = contents?.document;
    const win = contents?.window;
    if (!doc) return;

    view.applyStyles();
    attachTouchHandlers(view, doc, true, contents);
    attachInternalLinkInterceptor(view, doc);

    const checkIframeSelection = () => {
      try {
        const sel = win?.getSelection?.() || doc.getSelection?.();
        const raw = sel?.toString() || "";
        if (raw.trim().length > 0) {
          saveCapturedText(view, raw, undefined, contents, sel);
        }
      } catch {
        // Игнорируем
      }
    };

    let lastSelCheckTime = 0;
    let selThrottleTimer: number | null = null;

    const onThrottledSelectionChange = () => {
      const now = Date.now();
      const elapsed = now - lastSelCheckTime;
      if (elapsed >= SELECTION_THROTTLE_MS) {
        if (selThrottleTimer !== null) {
          view.clearSelectionTimer(selThrottleTimer);
          selThrottleTimer = null;
        }
        lastSelCheckTime = now;
        checkIframeSelection();
      } else if (selThrottleTimer === null) {
        selThrottleTimer = view.scheduleSelectionTimer(() => {
          selThrottleTimer = null;
          lastSelCheckTime = Date.now();
          checkIframeSelection();
        }, SELECTION_THROTTLE_MS - elapsed);
      }
    };

    const onDocTouchEnd = () => {
      view.recordReadingActivity();
      view.scheduleSelectionTimer(checkIframeSelection, 30);
      view.scheduleSelectionTimer(checkIframeSelection, 180);
    };

    const onDocMouseUp = () => {
      view.recordReadingActivity();
      view.scheduleSelectionTimer(checkIframeSelection, 20);
    };

    doc.addEventListener("selectionchange", onThrottledSelectionChange);
    doc.addEventListener("touchend", onDocTouchEnd, { passive: true });
    doc.addEventListener("mouseup", onDocMouseUp);

    view.cleanupListeners.push(() => {
      if (selThrottleTimer !== null) {
        view.clearSelectionTimer(selThrottleTimer);
        selThrottleTimer = null;
      }
      doc.removeEventListener("selectionchange", onThrottledSelectionChange);
      doc.removeEventListener("touchend", onDocTouchEnd);
      doc.removeEventListener("mouseup", onDocMouseUp);
    });
  });

  if (view.wrapEl) {
    attachTouchHandlers(view, view.wrapEl, false);
  }

  view.rendition.on("keyup", (e: KeyboardEvent) => {
    view.recordReadingActivity();
    if (e.key === "ArrowRight") view.turnPage("next");
    else if (e.key === "ArrowLeft") view.turnPage("prev");
    else if (e.key === "Escape" && view.isFullscreen) view.toggleFullscreen(false);
  });
}
